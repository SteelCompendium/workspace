type Sec = { text: string; lineStart: number; lineEnd: number };
const OUT = process.env.SC596_R3_OUT;
const rec = (name: string, obj: unknown) => {
	if (OUT) fs.appendFileSync(OUT, `${name}|${JSON.stringify(obj)}\n`);
};
const CTR = (n: number, name = 'Twin'): string[] => ['```ds-counter', `name: ${name}`, `current_value: ${n}`, '```'];
const bodyOf = (n: number, name = 'Twin'): string => `name: ${name}\ncurrent_value: ${n}`;
const pos = (line: number) => ({ line, col: 0, offset: 0 });
const settle = () => new Promise((r) => setTimeout(r, 25));
const noticeKind = (n: string): string =>
	n.includes('changed on disk first') ? 'dropped' : n.includes("can't be edited yet") ? 'unsavable-list' : n.includes("can't tell where") ? 'unsavable' : 'other:' + n.slice(0, 60);
const kinds = (): string[] => (Notice.notices as unknown[]).map((n) => noticeKind(String(n)));
function ctxOf(section: { current: Sec | null }, path = 'Note.md'): any {
	return { docId: 'doc-r3', sourcePath: path, frontmatter: undefined, addChild: () => {}, getSectionInfo: () => section.current };
}
function cacheOfText(text: string, omitHeadingAt: number[] = []) {
	const lines = text.split(/\r?\n/);
	const headings: Array<{ heading: string; level: number; position: unknown }> = [];
	const blocks: Record<string, { id: string; position: unknown }> = {};
	let fenced = false;
	lines.forEach((l, i) => {
		if (/^[> ]*```/.test(l)) fenced = !fenced;
		const h = !fenced && /^(#{1,6})\s+(.*?)\s*$/.exec(l);
		if (h && !omitHeadingAt.includes(i)) headings.push({ heading: h[2], level: h[1].length, position: { start: pos(i), end: pos(i) } });
		const id = /^\^(\w+)$/.exec(l);
		if (id) {
			let end = i - 1;
			while (end >= 0 && lines[end] === '') end--;
			let start = end;
			if (lines[end] === '```') for (start = end - 1; start >= 0 && !/^```\w/.test(lines[start]); start--);
			else if (/^>/.test(lines[end])) for (start = end; start > 0 && /^>/.test(lines[start - 1]); start--);
			blocks[id[1]] = { id: id[1], position: { start: pos(start), end: pos(end) } };
		}
	});
	return { headings, blocks };
}
function popoverEl(app: App, link: string | null): { popover: HTMLElement; embed: HTMLElement; el: HTMLElement } {
	const popover = document.createElement('div');
	popover.className = 'popover hover-popover';
	const embed = popover.appendChild(document.createElement('div'));
	embed.className = 'markdown-embed';
	const sub = link !== null && link.includes('#') ? link.slice(link.indexOf('#')) : null;
	if (sub !== null && sub.startsWith('#^')) embed.setAttribute('data-type', 'block');
	else if (sub !== null) {
		embed.setAttribute('data-type', 'heading');
		const h = embed.appendChild(document.createElement('h2'));
		h.setAttribute('data-heading', sub.split('#').pop()!.split('|')[0]);
	}
	const el = embed.appendChild(document.createElement('div'));
	document.body.appendChild(popover);
	if (link !== null) noteHoverLink(app, { linktext: link, sourcePath: 'Host.md', hoverParent: { hoverPopover: { hoverEl: popover } } });
	return { popover, embed, el };
}
function embedEl(src: string): HTMLElement {
	const embed = document.createElement('div');
	embed.className = 'internal-embed markdown-embed';
	embed.setAttribute('src', src);
	return embed.appendChild(document.createElement('div'));
}
function hostOf(app: App, el: HTMLElement, sec: Sec | null, body: string, path = 'Note.md') {
	const section = { current: sec };
	const host = new ReadingModeBlockHost(new Plugin(app) as never, el, ctxOf(section, path) as never, 'ds-counter');
	host.setMountedBody(body);
	return { host, section };
}
const formSave = (host: ReadingModeBlockHost, pin: any, body: string) => (pin ? host.replaceSource(body, { pin, pinOnly: true }) : host.replaceSource(body));
const names = (app: App): string[] => [...app.vault.getContent('Note.md')!.matchAll(/name: (\w+)/g)].map((m) => m[1]);
const values = (app: App): string[] => [...app.vault.getContent('Note.md')!.matchAll(/current_value: (\d+)/g)].map((m) => m[1]);
function freshApp(full: string, cache: () => unknown): App {
	const app = new App();
	app.vault.setFile('Note.md', full);
	jest.spyOn(app.metadataCache, 'getFileCache').mockImplementation(() => cache() as never);
	return app;
}
/** the text outside lines [a, a+n) of `text` */
const outside = (text: string, a: number, n: number) => {
	const l = text.split('\n');
	return [...l.slice(0, a), ...l.slice(a + n)].join('\n');
};

	test('CARD section gone at the click, NO twin', async () => {
