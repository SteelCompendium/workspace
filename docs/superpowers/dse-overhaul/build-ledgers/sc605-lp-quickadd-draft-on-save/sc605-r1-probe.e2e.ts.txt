const rec = (probe: string, step: string, data: unknown): void => fs.appendFileSync(OUT, JSON.stringify({ probe, step, data }) + '\n');

const TRACKER = 'Initiative/tracker.md';
const TWO = 'Initiative/two-trackers.md';

const LBL = 'init.malice.quickadd.label';

/** Page-side sampler: every animation frame AND every 4 ms, the quick-add boxes of every Live Preview tracker root (by root identity,
 *  leaf and index) and what holds focus; logs only changes. Also vault modifies and the prototype's trace (window.__dse605). */
const install = () =>
	dse.page(`(() => {
		const old = window.__zz605; if (old) { old.stop = true; clearInterval(old.timer); app.vault.offref(old.ref); }
		const ids = new WeakMap(); let next = 1;
		const rid = (r) => { if (!r) return null; if (!ids.has(r)) ids.set(r, next++); return ids.get(r); };
		const leaves = () => Array.from(document.querySelectorAll('.workspace-leaf'));
		const roots = () => Array.from(document.querySelectorAll('.markdown-source-view [data-dse-element="initiative"]')).filter((r) => !r.parentElement.closest('[data-dse-element]'));
		const t0 = performance.now();
		const T = () => (performance.now() - t0) | 0;
		const box = (r, id) => r.querySelector('[data-dse-control="' + id + '"]');
		const snapBoxes = () => roots().map((r) => { const leaf = leaves().indexOf(r.closest('.workspace-leaf')); const a = box(r, '${AMT}'); const l = box(r, '${LBL}'); return 'r' + rid(r) + '@L' + leaf + '=' + JSON.stringify([a ? a.value : null, l ? l.value : null]); }).join(' ');
		const desc = () => {
			const a = document.activeElement;
			if (!a || a === document.body) return 'body';
			const root = a.closest('[data-dse-element]');
			const ctl = a.closest('[data-dse-control]')?.getAttribute('data-dse-control');
			const what = ctl ?? (a.classList.contains('cm-content') ? 'cm-content' : a.tagName + '.' + String(a.className).slice(0, 30));
			const sel = (a.tagName === 'INPUT' && a.type === 'text') ? ' sel=' + a.selectionStart + '-' + a.selectionEnd : '';
			return what + '@r' + rid(root) + (a.matches(':focus-visible') ? '+ring' : '') + sel;
		};
		roots().forEach(rid);
		const z = { t0, log: [], lastB: null, lastF: null, desc, roots, rid, snapBoxes, leaves, box };
		const sample = (src) => { const b = snapBoxes(); if (b !== z.lastB) { z.log.push([T(), 'boxes', b, src]); z.lastB = b; } const d = desc(); if (d !== z.lastF) { z.log.push([T(), 'focus', d]); z.lastF = d; } };
		z.timer = setInterval(() => sample('int'), 4);
		const raf = () => { if (z.stop) return; sample('raf'); requestAnimationFrame(raf); }; requestAnimationFrame(raf);
		z.ref = app.vault.on('modify', (f) => z.log.push([T(), 'modify', f.path]));
		window.__zz605 = z; window.__dse605 = [];
		sample('start');
		return true;
	})()`);

/** The end state of every Live Preview tracker: boxes, hint, Add enabled, aria-invalid, round; plus focus. */
const endState = () =>
	dse.page(`(() => {
		const z = window.__zz605;
		const trackers = z.roots().map((r) => {
			const a = z.box(r, '${AMT}'); const l = z.box(r, '${LBL}'); const add = z.box(r, 'init.malice.quickadd.add');
			return {
				root: 'r' + z.rid(r), leaf: z.leaves().indexOf(r.closest('.workspace-leaf')),
				amount: a?.value ?? null, label: l?.value ?? null,
				hint: r.querySelector('.dse-init__malice-quickadd-hint')?.textContent ?? null,
				addEnabled: add ? !add.disabled : null,
				invalid: [a?.getAttribute('aria-invalid') ?? null, l?.getAttribute('aria-invalid') ?? null],
				placeholders: [a?.getAttribute('placeholder') ?? null, l?.getAttribute('placeholder') ?? null],
				round: r.querySelector('.dse-init__round-value')?.textContent ?? null,
			};
		});
		return { focus: z.desc(), trackers };
	})()`);

const finish = async (probe: string): Promise<void> => {
	await dse.settle();
	await browser.pause(1000);
	const end = await endState();
	const out = await dse.page(`(() => { const z = window.__zz605; z.stop = true; clearInterval(z.timer); app.vault.offref(z.ref); return { log: z.log, dbg: (window.__dse605 ?? []).map((e) => [Math.round(e[0] - z.t0), ...e.slice(1)]) }; })()`);
	rec(probe, 'result', { ...(out as object), end });
};
const mark = (probe: string, step: string) => dse.page(`window.__zz605.log.push([(performance.now() - window.__zz605.t0) | 0, 'mark', ${JSON.stringify(step)}])`).then(() => rec(probe, 'mark', step));

const draft = async (n: number, amount: string, label: string): Promise<void> => {
	const root = dse.block(INIT, n);
	await dse.type(root.control(AMT), amount, { replace: true });
	await dse.type(root.control(LBL), label, { replace: true });
};

		await mark('D', 'typed');
		await finish('D');
	});

	it('[ZZ605-D2] focus-in-box: mouse +, then click into the label, caret moved to the middle, inside the debounce', async () => {
		await dse.openNote(TRACKER, { mode: 'live-preview' });

		await dse.click(root.control(LBL));
		await browser.keys(['Home', 'ArrowRight', 'ArrowRight', 'ArrowRight']); // caret at 3
