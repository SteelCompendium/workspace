const rec = (probe: string, step: string, data: unknown): void => fs.appendFileSync(OUT, JSON.stringify({ probe, step, data }) + '\n');

const MIN = 'Initiative/tracker.md';

/** Every initiative root in every window (main + popouts): where it is, whether displayed, its boxes, round, malice; plus focus. */
const snap = () =>
	dse.page(`(() => {
		window.__zzr3ids = window.__zzr3ids || new WeakMap(); window.__zzr3n = window.__zzr3n || 1;
		const rid = (r) => { if (!window.__zzr3ids.has(r)) window.__zzr3ids.set(r, window.__zzr3n++); return window.__zzr3ids.get(r); };
		const docs = [document];
		for (const l of app.workspace.getLeavesOfType('markdown')) { const d = l.view.containerEl.ownerDocument; if (!docs.includes(d)) docs.push(d); }
		const out = [];
		docs.forEach((doc, di) => {
			const leaves = Array.from(doc.querySelectorAll('.workspace-leaf'));
			for (const r of doc.querySelectorAll('[data-dse-element="initiative"]')) {
				if (r.parentElement.closest('[data-dse-element]')) continue;
				const q = (id) => r.querySelector('[data-dse-control="' + id + '"]');
				out.push({
					root: 'r' + rid(r), doc: di, leaf: leaves.indexOf(r.closest('.workspace-leaf')),
					where: r.closest('.cm-content') ? 'cm' : r.closest('.markdown-reading-view') ? 'reading' : 'other',
					embed: !!r.closest('.internal-embed, .markdown-embed'),
					shown: r.offsetParent !== null,
					amount: q('${AMT}')?.value ?? null, label: q('${LBL}')?.value ?? null,
					hint: r.querySelector('.dse-init__malice-quickadd-hint')?.textContent ?? null,
					invalid: !!(q('${AMT}')?.hasAttribute('aria-invalid') || q('${LBL}')?.hasAttribute('aria-invalid')),
					round: r.querySelector('.dse-init__round-value')?.textContent ?? null,
					malice: r.querySelector('.dse-init__malice-value')?.textContent ?? null,
				});
			}
		});
		const focus = docs.map((doc) => { const a = doc.activeElement; if (!a || a === doc.body) return 'body'; const c = a.closest('[data-dse-control]')?.getAttribute('data-dse-control'); return (c ?? (a.classList.contains('cm-content') ? 'cm-content' : a.tagName)) + (a.matches(':focus-visible') ? '+ring' : '') + (a.tagName === 'INPUT' && a.type === 'text' ? ' sel=' + a.selectionStart + '-' + a.selectionEnd : ''); });
		return { roots: out, focus };
	})()`);

const settled = async (): Promise<void> => {
	await dse.settle();
	await browser.pause(1000);
};
const fill = async (n: number, amount: string, label: string): Promise<void> => {
	const root = dse.block(INIT, n);
	await dse.type(root.control(AMT), amount, { replace: true });
	await dse.type(root.control(LBL), label, { replace: true });
};
