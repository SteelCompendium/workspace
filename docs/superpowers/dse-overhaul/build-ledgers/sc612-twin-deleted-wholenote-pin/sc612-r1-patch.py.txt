/** SC-612 r1 probe: one greedy extreme ('head': only the common head trimmed, the forward diff over the rest; 'tail': the mirror). Null: changed or removed; 'gave up': more than LINE_MAP_MAX_EDITS. */
export function mapLineGreedy(before: string, after: string, line: number, dir: 'head' | 'tail'): number | null | 'gave up' {
	if (before === after) return line;
	const a = splitLines(before);
	const b = splitLines(after);

/** SC-612 r1 probe: the two GREEDY extremes - head-first trims only the common head and runs the forward diff over the rest (the earliest
 * pairing), tail-first trims only the common tail and runs the diff over the reversed rest (the latest pairing). A tie between identical
 * blocks is seen even when an unrelated edit elsewhere stops one of the trims. */
export function mapLineStrictPure(before: string, after: string, line: number): StrictLine {
	if (before === after) return { line };
	const a = splitLines(before);
	const b = splitLines(after);
	if (line < 0 || line >= a.length) return 'deleted';
	const head = mapLineOf(a, b, line, false);
	const r = mapLineOf(a.slice().reverse(), b.slice().reverse(), a.length - 1 - line, false);
	const tail = r === null || r === GAVE_UP ? r : b.length - 1 - r;
	if (head === GAVE_UP || tail === GAVE_UP) return 'gave up';
	if (head === null && tail === null) return 'deleted';
	return head !== null && head === tail ? { line: head } : 'disagree';
}
