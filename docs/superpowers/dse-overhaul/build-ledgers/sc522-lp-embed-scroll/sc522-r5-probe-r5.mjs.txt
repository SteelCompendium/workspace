function r3Kit(t) {
	const A = 'window.__r3A';
	const mouse = (type, x, y, extra = {}) => t.cdp.call('Input.dispatchMouseEvent', { type, x, y, ...extra });
	const pressAt = async ({ x, y }, count = 1) => {
		await mouse('mouseMoved', x, y);
		for (let c = 1; c <= count; c++) {
			await mouse('mousePressed', x, y, { button: 'left', buttons: 1, clickCount: c });
			await mouse('mouseReleased', x, y, { button: 'left', buttons: 0, clickCount: c });
		}
	};
	const ctrlA = async () => {
		await t.cdp.call('Input.dispatchKeyEvent', { type: 'rawKeyDown', key: 'a', code: 'KeyA', windowsVirtualKeyCode: 65, nativeVirtualKeyCode: 65, modifiers: 2 });
		await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyUp', key: 'a', code: 'KeyA', windowsVirtualKeyCode: 65, nativeVirtualKeyCode: 65, modifiers: 2 });
	};
	const snap = () => t.ev(`window.__r3.snap(${A})`);
	const openHost = async (rel, mode = 'lp', viaOther = true) => {
		if (viaOther) await t.open('Lifecycle/other.md');
		else await t.open(rel); // G-S52's openHost: the leaf goes to the host's Reading view first (the host is never closed)
		const st = mode === 'lp' ? { mode: 'source', source: false } : { mode: 'preview' };
		await t.ev(`(async () => { window.__r3A = app.workspace.getMostRecentLeaf(); await window.__r3A.setViewState({ type: 'markdown', state: Object.assign({ file: ${JSON.stringify(rel)} }, ${JSON.stringify(st)}), active: true }); if (window.__r3A.view.editor && ${mode === 'lp'}) window.__r3A.view.editor.setCursor({ line: 0, ch: 0 }); return true; })()`);
		await t.sleep(1500);
	};
	const scroller = `window.__r3.cm(${A}).scrollDOM`;
	// park relative to the i-th DSE embed: end | below | in | view | top:<px from embed top> ; or el:<expr> (element centred)
	const park = async (where, i = 0) => {
		await t.ev(`(() => { ${scroller}.scrollTop = 0; return true; })()`);
		for (let k = 0; k < 80 && !(await t.ev(`(() => { const e = window.__r3.emb(${A}, ${i}); return !!e && !!e.querySelector('[data-dse-element]') && e.querySelector('[data-dse-element]').childElementCount > 0; })()`)); k++) {

		}

			await t.ev(`(() => { const s = ${scroller}; const e = window.__r3.emb(${A}, ${i}); const sr = s.getBoundingClientRect(); const top = e.getBoundingClientRect().top - sr.top + s.scrollTop; const h = e.getBoundingClientRect().height; const w = ${JSON.stringify(where)};

						await t.reset(R3_B);
						await k.openHost(R3_TALL, 'lp', false);
					} else if (kase === 'gs52seq') {
						// G-S52 (d) then (e) without its fresh write: add a hero with the host parked below, reset B, reopen the host
						await k.park('below');
						await t.edit(R3_B, `c.replace('enemy_groups: []', ${JSON.stringify(`  - name: "Added ${Date.now() % 100000}"\n    max_stamina: 30\nenemy_groups: []`)})`);

						await k.openHost(R3_TALL);

						// a click lands; the host is closed (another note in the leaf); B is reset (in-app or on disk); the host reopened
						await k.park('in');
						const at = await t.ev(k.visible(EMB, 'button.dse-init__portrait-toggle'));
						const b0 = t.read(R3_B);
						await k.pressAt(at);
						for (let i = 0; i < 40 && t.read(R3_B) === b0; i++) await t.sleep(100);

						await t.sleep(800);
						await t.open('Lifecycle/other.md');
						if (kase === 'closed-vault') await t.reset(R3_B);
						else { fs.writeFileSync(path.join(vault, R3_B), FIXTURES[R3_B]); await t.sleep(1500); }
						await k.openHost(R3_TALL);
					}
					await k.park('in');
					await t.sleep(500);
					const disk = t.read(R3_B);
					row.diskTurns = (disk.match(/has_taken_turn: true/g) || []).length;
					row.diskHeroes = (disk.match(/- name:/g) || []).length;
					row.embHeroes = await t.ev(`(() => { const e = ${EMB}; return e ? e.querySelectorAll('.dse-init__name').length : null; })()`);
					row.embTurnsPressed = await t.ev(`(() => { const e = ${EMB}; return e ? e.querySelectorAll('button.dse-init__portrait-toggle[aria-pressed="true"]').length : null; })()`);
					row.embMalice = await t.ev(`(() => { const e = ${EMB}; const m = e && e.querySelector('.dse-init__malice-num, .dse-init__malice-value'); return m ? m.textContent : null; })()`);

					const at = await t.ev(k.visible(EMB, 'button.dse-init__portrait-toggle'));
					const before = t.read(R3_B);
					await k.pressAt(at);
					for (let i = 0; i < 40 && t.read(R3_B) === before; i++) await t.sleep(100);
					row.landed = t.read(R3_B) !== before;

					row.notices = await t.noticesSince(m0);
