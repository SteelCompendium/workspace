#!/bin/bash
# r3 probe.sh <tag> <bundle-name> <only-ids> [env assignments...]: scratch harness copy (probe block injected) against a scratch bundle.
S=/tmp/claude-1000/-home-scott-code-steelCompendium-workspace/be71a0a9-199f-4e14-967b-191935312b05/scratchpad/sc522-r3-a79dd8ba57f5042f0
W=/home/scott/code/steelCompendium/worktrees/sc522-lp-embed-scroll/draw-steel-elements
T=$1; B=$2; ONLY=$3; shift 3
H="$S/harness-$T.mjs"
mkdir -p "$S/logs" "$S/out"
node "$S/mkharness.cjs" "$W/visual-harness/obsidian-lifecycle.mjs" "$H" "$S/probe-r3.mjs.txt" > "$S/logs/mk-$T.log" 2>&1 || { echo mk-failed; cat "$S/logs/mk-$T.log"; exit 9; }
LOG="$S/logs/probe-$T.log"
echo "bundle $B main.js $(sha256sum $S/bundles/$B/main.js | cut -c1-12)" > "$LOG"
env "$@" S522_REPO="$W" R3_OUT="$S/out" R3_VAR="$T" DSE_LIFECYCLE_PORT=${R3_PORT:-9573} DSE_LIFECYCLE_BUNDLE="$S/bundles/$B" timeout 595 node "$H" --only="$ONLY" >> "$LOG" 2>&1
rc=$?
echo "rc=$rc" >> "$LOG"
echo "porcelain-after: [$(git -C "$W" status --porcelain | tr '\n' ' ')] head: $(git -C "$W" rev-parse --short HEAD)" >> "$LOG"
grep -E "OBSIDIAN-LIFECYCLE (start|done|temp|environment)|FAIL" "$LOG" | cut -c1-600
echo "rc=$rc log=$LOG"
exit $rc
