FILES = ['test/dom/framework/livePreviewScrollHold.test.ts', 'test/dom/framework/pipeline.test.ts', 'test/dom/framework/previewScrollPin.test.ts', 'test/dom/framework/register-framework-elements.test.ts']
def git(*a): return subprocess.run(['git', '-C', W, *a], capture_output=True, text=True)
def jest(tag):
    out = f'{S}/logs/mut-{tag}.json'
    subprocess.run(['npx', 'jest', *FILES, '--json', f'--outputFile={out}'], cwd=W, capture_output=True, text=True)
    try:
        j = json.load(open(out))
    except Exception as e:
        return None, [f'jest json unreadable: {e}']
    failed = [a['fullName'] for t in j['testResults'] for a in t['assertionResults'] if a['status'] == 'failed']
    # a suite that failed to compile has no assertion results
    bad_suites = [t['name'] for t in j['testResults'] if t['status'] == 'failed' and not any(a['status']=='failed' for a in t['assertionResults'])]
    return j['numPassedTests'], failed + [f'SUITE-ERROR {b}' for b in bad_suites]
rows = []
killed = 0
assert git('status', '--porcelain').stdout.strip() == '', 'tree must be clean'
for i, (name, f, old, new) in enumerate(M, 1):
    p = os.path.join(W, f)
    s = open(p).read()
    if s.count(old) != 1:
        rows.append((i, name, 'NOT-APPLIED (old text count %d)' % s.count(old), [])); continue
    try:
        open(p, 'w').write(s.replace(old, new))
        passed, failed = jest(str(i))
    finally:
        git('checkout', '--', f)
    ok = bool(failed)
    killed += ok
    rows.append((i, name, 'KILLED' if ok else 'SURVIVED', failed))
assert git('status', '--porcelain').stdout.strip() == '', 'tree not restored'
with open(f'{S}/logs/mutation-summary.txt', 'w') as o:

        o.write(f'{r[0]:2d} {r[2]:9s} {r[1]}\n')

    o.write(f'KILLED {killed} / {len(M)}\n')
print(f'KILLED {killed} / {len(M)}')
