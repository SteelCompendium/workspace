	async run(t) {
		await t.ev(P522_INSTALL);
		await t.ev(`window.__p522.moLayout = ${process.env.P522_MOLAYOUT === '1'}`);
		await t.ev(`window.__p522.trace = ${process.env.P522_TRACE === '1'}`);

		const reps = Number(process.env.P522_REPS ?? 2);
		const cases = (process.env.P522_CASES ?? 'process:end:E').split(',');
		const prev = await t.ev(`${t.reg}.enabled`);
		const rows = [];
		try {
			await t.ev(`${t.reg}.enabled = ${adopt === 'on' ? 'true' : 'false'}`);
			for (const kase of cases) {
				for (let rep = 0; rep < reps; rep++) {
					try {
						rows.push(await p522Case(t, kase, rep, adopt));
					} catch (e) {
						rows.push({ kase, rep, adopt, error: String((e && e.stack) || e) });
						await t.ev(`(() => { document.querySelectorAll('.modal-container .modal-close-button').forEach((b) => b.click()); return true; })()`);
					} finally {
						await t.ev(`(() => { if (window.__p522B) { try { window.__p522B.detach(); } catch (e) {} window.__p522B = null; } return true; })()`);
						for (const sh of new Set(Object.values(P522_SHAPES).map((x) => x.b))) if (t.read(sh) !== FIXTURES[sh]) await t.reset(sh);
						await t.sleep(300);
					}
				}
			}
		} finally {
			await t.ev(`${t.reg}.enabled = ${JSON.stringify(prev)}`);
			await t.open('Lifecycle/other.md');
		}
		p522Write(`REPRO-${P522_VAR}-${adopt}`, rows);
		return '\n  ' + rows.map((r) => (r.error ? `${r.kase}#${r.rep}: ERROR ${r.error.split('\n').slice(0, 2).join(' / ')}` : p522Line(r))).join('\n  ');
	},
});
