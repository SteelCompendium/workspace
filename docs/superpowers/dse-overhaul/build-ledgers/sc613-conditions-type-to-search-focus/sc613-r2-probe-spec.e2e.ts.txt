/** A window-BUBBLE keydown recorder (runs after every document listener): what each real key's target was and whether it ended cancelled. */
const installRecorder = (): Promise<unknown> =>
	dse.page(`(() => {
		window.__rvwKeys = [];
		if (!window.__rvwRec) {
			window.__rvwRec = (e) => window.__rvwKeys.push({ key: e.key, target: e.target === e.target.ownerDocument?.body ? 'body' : e.target === e.target.ownerDocument?.documentElement ? 'html' : ((e.target.getAttribute && (e.target.getAttribute('data-dse-control') || e.target.className)) || e.target.tagName), dp: e.defaultPrevented });
			window.addEventListener('keydown', window.__rvwRec);
		}
		return true;
	})()`);
const keys = (): Promise<unknown> => dse.page(`window.__rvwKeys`);

const state = (): Promise<Record<string, unknown>> =>
	dse.page(`(() => {
		const inp = document.querySelector('.modal-container .dse-condal__input');
		const a = document.activeElement;
		const mc = Array.from(document.querySelectorAll('.modal-container'));
		return {
			modals: mc.length,
			order: mc.map((m) => m.querySelector('.dse-condal-modal') ? 'conditions' : m.querySelector('.prompt') ? 'prompt' : m.querySelector('.dse-sedit__row') ? 'stamina' : (m.className + '|' + (m.firstElementChild ? m.firstElementChild.className : ''))),
			input: inp ? inp.value : null,
			focus: a === document.body ? 'body' : a ? (a.getAttribute('data-dse-control') || a.tagName + '.' + a.className) : null,
			menus: document.querySelectorAll('.menu').length,
			suggests: document.querySelectorAll('.suggestion-container').length,
		};
	})()`);

/** Which .modal-container is hit-tested at the centre of the viewport (the visually top-most). */
const topAtCentre = (): Promise<unknown> =>
	dse.page(`(() => {
		const el = document.elementFromPoint(innerWidth / 2, innerHeight / 2);
		const mc = el && el.closest('.modal-container');
		const all = Array.from(document.querySelectorAll('.modal-container'));
		return { index: mc ? all.indexOf(mc) : -1, of: all.length, zs: all.map((m) => getComputedStyle(m).zIndex), hit: el ? el.className : null };
	})()`);

const clickTitle = (): Promise<void> => $('.modal-container .dse-condal-modal').$('..').$('.dse-modal__title').click();
const blurToBody = (): Promise<unknown> => dse.page(`(() => { document.activeElement && document.activeElement.blur && document.activeElement.blur(); return document.activeElement === document.body; })()`);
