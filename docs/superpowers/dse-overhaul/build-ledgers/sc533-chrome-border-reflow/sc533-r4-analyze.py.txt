#!/usr/bin/env python3
# SC-533 r2b: aggregate the final-vs-stock-vs-stub sessions (c, d). usage: analyze-r2b.py <out dir>
import json, statistics, csv, sys
SC = '/tmp/claude-1000/-home-scott-code-steelCompendium-workspace/be71a0a9-199f-4e14-967b-191935312b05/scratchpad/sc533-r4-08f91447'
OUT = sys.argv[1]
rows, meta = [], {}
for tag in 'ab':
    for scene in ['R', 'LP']:
        d = json.load(open(f'{SC}/out/sc533-r4-raw-{tag}-{scene}.json'))
        meta[f'{tag}-{scene}'] = dict(loadStart=d['meta']['loadStart'], loadEnd=d['meta']['loadEnd'], obsidian=d['meta']['obsidian'], roff=d['meta']['roff'])
        for r in d['reps']:
            rows.append(dict(session=f'{tag}-{scene}', scene=scene, seq=r['seq'], round=r['round'], variant=r['variant'], warmup=int(r['warmup']), k=r['k'],
                             tFound=r['tFound'], tRaf=r['tRaf'], tPost=r['tPost'], tRaf2=r['tRaf2'], tProc=r['tProc'], timedOut=int(r['timedOut']), load=r['load']))
with open(f'{OUT}/sc533-r4-raw-draws.csv', 'w', newline='') as f:
    w = csv.DictWriter(f, fieldnames=list(rows[0].keys())); w.writeheader(); w.writerows(rows)
def q(xs, p):
    xs = sorted(xs); pos = (len(xs) - 1) * p; lo = int(pos); hi = min(lo + 1, len(xs) - 1)
    return xs[lo] + (xs[hi] - xs[lo]) * (pos - lo)
def st(xs):
    return dict(n=len(xs), med=round(statistics.median(xs), 1), p90=round(q(xs, .9), 1), iqr=round(q(xs, .75) - q(xs, .25), 1), min=round(min(xs), 1), max=round(max(xs), 1))
res = {'meta': meta}
V = ['stock', 'final', 'stub']
for scene in ['R', 'LP']:
    m = res[scene] = {}
    sel = [r for r in rows if r['scene'] == scene and not r['timedOut']]
    for v in V:
        t = [r for r in sel if r['variant'] == v and not r['warmup']]
        m[v] = dict(tRaf=st([r['tRaf'] for r in t]), tPost=st([r['tPost'] for r in t]))
    keys = sorted(set((r['session'], r['round']) for r in sel))
    vis = lambda s, rd, v: statistics.median([r['tPost'] for r in sel if r['session'] == s and r['round'] == rd and r['variant'] == v and not r['warmup']])
    for a, b in [('final', 'stub'), ('stock', 'final'), ('stock', 'stub')]:
        diffs = [round(vis(s, rd, a) - vis(s, rd, b), 1) for s, rd in keys]
        m[f'paired_{a}_minus_{b}'] = dict(n=len(diffs), med=round(statistics.median(diffs), 1), all=diffs)
    u = m['stub']['tPost']['med']; bar = max(50, .1 * u); dd = m['final']['tPost']['med'] - u
    m['bar'] = dict(stub_med=u, bar_ms=round(bar, 1), final_minus_stub_med=round(dd, 1), PASS=bool(dd <= bar))
json.dump(res, open(f'{OUT}/sc533-r4-stats.json', 'w'), indent=1)
print(json.dumps(meta))
for scene in ['R', 'LP']:
    for v in V: print(scene, v, m_ := res[scene][v]['tRaf'], res[scene][v]['tPost'])
    for k, val in res[scene].items():
        if k.startswith('paired') or k == 'bar': print(scene, k, val)
print('rows', len(rows), 'timedOut', sum(r['timedOut'] for r in rows))
