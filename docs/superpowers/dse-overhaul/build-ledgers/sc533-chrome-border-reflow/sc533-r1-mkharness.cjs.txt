// SC-533 r1 (adapted from SC-522 r1's mkharness): build a SCRATCH copy of the lifecycle harness with probe blocks
// injected before main(). The copy resolves `repo` (demo-vault) from $S533_REPO. No tree is edited.
// usage: node mkharness.cjs <harness source> <out file> <probe block file>...
const fs = require('fs');
const [src, out, ...blocks] = process.argv.slice(2);
let s = fs.readFileSync(src, 'utf8');
const repoLine = 'const repo = path.dirname(dir);';
if (!s.includes(repoLine)) { console.error('repo line not found'); process.exit(3); }
s = s.replace(repoLine, 'const repo = process.env.S533_REPO ?? path.dirname(dir);');
const anchor = '\nasync function main() {';
if (!s.includes(anchor)) { console.error('main anchor not found'); process.exit(3); }
const blk = blocks.map((b) => fs.readFileSync(b, 'utf8')).join('\n');
s = s.replace(anchor, '\n' + blk + '\n' + anchor);
fs.writeFileSync(out, s);
console.log(`wrote ${out} (${s.length} bytes, ${blocks.length} block(s))`);
