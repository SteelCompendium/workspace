const R3X_INSTALL = `(() => {
	window.__r3install = (w, tag) => {
		if (w.__r3) return true;
		const L = (w.__r3 = { tag, reqs: [], log: [], depth: 0, cur: null, seq: 0, hold: false, held: [] });
		const rRaf = w.requestAnimationFrame.bind(w);
		const rCaf = w.cancelAnimationFrame.bind(w);
		w.requestAnimationFrame = function (cb) {
			const stack = new Error().stack || '';
			const plugin = stack.includes('plugin:draw-steel-elements');
			const rec = { id: ++L.seq, at: performance.now(), inRaf: L.depth > 0, plugin, ran: null, reads: 0, writes: 0, cancelled: false, held: false, h: 0 };
			L.reqs.push(rec);
			if (L.reqs.length > 40000) L.reqs.splice(0, 20000);
			const run = (ts) => { L.depth++; const prev = L.cur; L.cur = rec; rec.ran = performance.now(); try { return cb(ts); } finally { L.cur = prev; L.depth--; } };
			if (L.hold && plugin) { rec.held = true; rec.h = -rec.id; L.held.push({ fake: rec.h, run, rec }); return rec.h; }
			rec.h = rRaf(run);
			return rec.h;
		};
		w.cancelAnimationFrame = function (h) {
			if (h < 0) { const i = L.held.findIndex((x) => x.fake === h); if (i >= 0) { L.held[i].rec.cancelled = true; L.held.splice(i, 1); } return undefined; }
			for (let i = L.reqs.length - 1; i >= 0; i--) if (L.reqs[i].h === h) { L.reqs[i].cancelled = true; break; }
			return rCaf(h);
		};
		L.release = () => { const hs = L.held.splice(0); for (const x of hs) { x.rec.held = 'released'; rRaf(x.run); } return hs.length; };
		const rGcs = w.getComputedStyle;
		L.gcs = (el) => rGcs.call(w, el);
		w.getComputedStyle = function (el, p) {
			if (el && el.classList && el.classList.contains('dse-chrome-anchor')) { L.log.push(['R', performance.now(), L.cur ? L.cur.id : 0, el.isConnected ? 1 : 0]); if (L.cur) L.cur.reads++; }
			return rGcs.call(this, el, p);
		};
		const proto = w.CSSStyleDeclaration.prototype;
		const rSet = proto.setProperty;
		proto.setProperty = function (prop, v, pr) {
			if (typeof prop === 'string' && prop.startsWith('--dse-chrome-frame-border')) { L.log.push(['W', performance.now(), L.cur ? L.cur.id : 0, v]); if (L.cur) L.cur.writes++; }
			return rSet.call(this, prop, v, pr);
		};
		L.mark = () => ({ r: L.seq, l: L.log.length });
		/** Plugin rAF flushes since mark m: reads/writes, inRaf, order (all R before W). */
		L.flushes = (m) => L.reqs.filter((r) => r.id > m.r && r.plugin && (r.reads > 0 || r.writes > 0)).map((r) => {
			const seq = L.log.filter((e) => e[2] === r.id).map((e) => e[0]).join('');
			return { id: r.id, inRaf: r.inRaf, wait: r.ran === null ? null : +(r.ran - r.at).toFixed(1), reads: r.reads, writes: r.writes, entries: r.writes / 2, orderOk: /^R*W*$/.test(seq), held: r.held };
		});
		L.pluginReqs = (m) => L.reqs.filter((r) => r.id > m.r && r.plugin).length;
		/** Every anchor read/write since mark that happened OUTSIDE a rAF callback (i.e. in a mount task). */
		L.outside = (m) => L.log.slice(m.l).filter((e) => e[2] === 0).length;
		/** Geometry + vars of every connected, displayed panel under scope. */
		L.geo = (scope) => Array.from(scope.querySelectorAll('.dse-chrome')).filter((p) => p.isConnected).map((p) => {
			const a = p.parentElement;
			const cs = L.gcs(a);
			const pcs = L.gcs(p);
			const pr = p.getBoundingClientRect();
			const ar = a.getBoundingClientRect();
			const bt = parseFloat(cs.borderTopWidth) || 0;
			const br = parseFloat(cs.borderRightWidth) || 0;
			const ct = pcs.getPropertyValue('--dse-chrome-frame-border-top').trim();
			const cr = pcs.getPropertyValue('--dse-chrome-frame-border-right').trim();
			const shown = pcs.display !== 'none' && pr.width > 0;
			const varOk = (bt === 0 ? ct === '' || ct === '0px' : ct === bt + 'px') && (br === 0 ? cr === '' || cr === '0px' : cr === br + 'px');
			return { el: p.closest('[data-dse-element]')?.getAttribute('data-dse-element'), bt, br, ct, cr, inline: p.style.getPropertyValue('--dse-chrome-frame-border-top'), anchorInline: a.style.getPropertyValue('--dse-chrome-frame-border-top'), shown, insetR: +(ar.right - pr.right).toFixed(2), gapTop: +(ar.top - pr.bottom).toFixed(2), varOk };
		});
		return true;
	};
	return window.__r3install(window, 'main');
})()`;
