#!/usr/bin/env bash
# SC-533 r2b: build stock / final (branch HEAD) / stub bundles from the scratch export (never the worktree).
set -u
SC=/tmp/claude-1000/-home-scott-code-steelCompendium-workspace/be71a0a9-199f-4e14-967b-191935312b05/scratchpad/sc533-r2-7dca00e8
D=/home/scott/code/steelCompendium/worktrees/sc533-chrome-border-reflow/draw-steel-elements
T=$SC/src
F=$T/src/framework/chrome/mountChrome.ts
mkdir -p "$SC/variants-r2b" "$SC/bundles"
git -C "$D" show 4deb8cf:src/framework/chrome/mountChrome.ts > "$SC/variants-r2b/stock.ts"
git -C "$D" show HEAD:src/framework/chrome/mountChrome.ts > "$SC/variants-r2b/final.ts"
cp "$SC/variants-src/stub/mountChrome.ts" "$SC/variants-r2b/stub.ts"
for V in stock final stub; do
  cp "$SC/variants-r2b/$V.ts" "$F"
  ( cd "$T" && node esbuild.config.mjs production ) > "$SC/build-$V-r2b.log" 2>&1
  echo "build $V rc=$?"
  mkdir -p "$SC/bundles/$V"
  cp "$T/main.js" "$T/styles.css" "$T/manifest.json" "$SC/bundles/$V/"
  printf '\n;globalThis.__sc533variant = "%s";\n' "$V" >> "$SC/bundles/$V/main.js"
  diff -u "$SC/variants-r2b/stock.ts" "$F" > "$SC/sc533-r2b-variant-$V.patch"
done
cp "$SC/variants-r2b/stock.ts" "$F"
sha256sum "$SC"/bundles/stock/main.js "$SC"/bundles/final/main.js "$SC"/bundles/stub/main.js "$SC"/bundles/final/styles.css | sed "s#$SC/##"
git -C "$D" rev-parse --short HEAD
