def mods_without(what, src_line):
    full = ['evt.ctrlKey', 'evt.metaKey', 'evt.altKey', 'evt.repeat', 'evt.defaultPrevented']
    return src_line.replace(' || '.join(full), ' || '.join(p for p in full if p != what))

def add(mid, desc, file, old, new, tests):
    MUTANTS.append((mid, desc, file, old, new, tests))

def sh(*a, **k):
    return subprocess.run(a, capture_output=True, text=True, **k)

def main():
    log, which = sys.argv[1], sys.argv[2]
    want = None if which == 'all' else set(which.split(','))
    out = open(log, 'a')
    def w(s):
        out.write(s + '\n'); out.flush()
    for mid, desc, file, old, new, tests in MUTANTS:
        if want is not None and mid not in want:
            continue
        path = os.path.join(W, file)
        orig = open(path).read()
        if old == '@BASE@':
            r = sh('git', '-C', W, 'checkout', 'e5b6f9f2', '--', file); assert r.returncode == 0, r.stderr
        elif old == '@BASE597@':
            r = sh('git', '-C', W, 'checkout', '94133a5a', '--', file); assert r.returncode == 0, r.stderr
        else:
            assert orig.count(old) == 1, (mid, old, orig.count(old))
            open(path, 'w').write(orig.replace(old, new))
        r = sh('npx', 'jest', *tests, '--silent', cwd=W)
        text = r.stdout + r.stderr
        fails = []
        for line in text.splitlines():
            m = re.match(r'^\s+● (.+)$', line)
            if m and m.group(1) not in fails:
                fails.append(m.group(1))
        summ = [l for l in text.splitlines() if l.startswith('Tests:')]
        sh('git', '-C', W, 'checkout', 'HEAD', '--', file)
        d = sh('git', '-C', W, 'diff', '--stat').stdout.strip()
        w(f'## {mid}: {desc}')
        w(f'file: {file}')
        w('command: npx jest ' + ' '.join(tests) + ' --silent')
        w(f'result: {"KILLED" if r.returncode != 0 else "SURVIVED"} (jest rc {r.returncode}); {summ[0] if summ else "no Tests line"}')
        for f in fails:
            w('  red: ' + f)
        w(f'restored: git diff --stat empty = {d == ""}' + ('' if d == '' else ' !!! ' + d))
        w('')
        assert d == '', d
    out.close()

main()
