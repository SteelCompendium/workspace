#!/bin/bash
# usage: run-lifecycle.sh <bundle dir or -> <only ids> <logfile> <camtmp dir>
# The WORKTREE's own harness. A bundle dir pins another bundle (dse-verify G-S83 footgun 9); "-" builds the worktree head.
if [ "$1" != "-" ]; then export DSE_LIFECYCLE_BUNDLE="$1"; fi
export DSE_CAMERA_TMP="$4"
mkdir -p "$4"
cd /home/scott/code/steelCompendium/worktrees/sc598-hover-modal-fast-second-click/draw-steel-elements || exit 9
start=$(date +%s)
if [ "$1" = "-" ]; then npm run obsidian-lifecycle -- --only="$2" > "$3" 2>&1; else node visual-harness/obsidian-lifecycle.mjs --only="$2" > "$3" 2>&1; fi
rc=$?
end=$(date +%s)
echo "LIFECYCLE_RC=$rc WALL_S=$((end-start)) BUNDLE=$1 LOAD=$(cat /proc/loadavg)" >> "$3"
exit $rc
