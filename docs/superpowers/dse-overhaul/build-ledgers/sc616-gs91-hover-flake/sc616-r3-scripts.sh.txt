######## run.sh
#!/bin/bash
# usage: run.sh <tag> <new|old> <bundle: head|DIRNAME> <count> [VAR=val ...]
#  new = the worktree's harness (33dd0a1b); old = DIRNAME's own harness (git archive 1516f894) with its own bundle
#  each run: logs/<tag>-<i>-<stamp>.log with a trailing rc line; index line appended to logs/index.txt
S=/tmp/claude-1000/-home-scott-code-steelCompendium-workspace/be71a0a9-199f-4e14-967b-191935312b05/scratchpad/sc616-r3-rv
W=/home/scott/code/steelCompendium/worktrees/sc616-gs91-hover-flake/draw-steel-elements
tag=$1; which=$2; bundle=$3; n=$4
shift 4
for kv in "$@"; do export "$kv"; done
if [ "$which" = new ]; then
  H=$W/visual-harness/obsidian-lifecycle.mjs
  cd $W || exit 9
  if [ "$bundle" != head ]; then export DSE_LIFECYCLE_BUNDLE=$S/$bundle; fi
else
  H=$S/$bundle/visual-harness/obsidian-lifecycle.mjs
  cd $S/$bundle || exit 9
fi
for i in $(seq 1 $n); do
  log="$S/logs/$tag-$i-$(date +%H%M%S%N | cut -c1-9).log"
  echo "load $(cat /proc/loadavg) harness=$which bundle=$bundle H=$H BUNDLE=$DSE_LIFECYCLE_BUNDLE env: $*" > "$log"
  start=$(date +%s)
  node $H --only=${ONLY_IDS:-G-S91} >> "$log" 2>&1
  rc=$?
  end=$(date +%s)
  echo "lc_rc=$rc wall=$((end-start))s load_end $(cat /proc/loadavg)" >> "$log"
  echo "$tag-$i rc=$rc wall=$((end-start)) log=$log" >> "$S/logs/index.txt"
done
