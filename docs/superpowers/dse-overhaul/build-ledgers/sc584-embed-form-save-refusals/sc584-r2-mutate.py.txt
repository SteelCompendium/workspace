    'mA': ("if (!fence || normalizeBody(fence.body) !== normalizeBody(known) || !sameAt(fence.lineEnd + 1, after)) continue;",

    'mE': ("			if (found >= 0) return null;\n			found = fence.lineStart;",

           "this.openTimeBodyLine(whole, sectionText, pin.lineStart, this.knownBody);"),

name = sys.argv[1]
src = open(F).read()
