/** Install a page-side sampler: every 5 ms record what holds focus (control id @ root id, ring), plus focus events and vault modifies. */
const install = (el: string, view = '.markdown-source-view') =>
	dse.page(`(() => {
		if (window.__zz602?.timer) clearInterval(window.__zz602.timer);
		if (window.__zz602?.ref) app.vault.offref(window.__zz602.ref);
		const ids = new WeakMap(); let next = 1;
		const rid = (r) => { if (!r) return null; if (!ids.has(r)) ids.set(r, next++); return ids.get(r); };
		const sel = '.workspace-leaf.mod-active ${view} [data-dse-element="${el}"]';
		const roots = () => Array.from(document.querySelectorAll(sel)).filter((r) => !r.parentElement.closest('[data-dse-element]'));
		const t0 = performance.now();
		const T = () => (performance.now() - t0) | 0;
		const desc = () => {
			const a = document.activeElement;
			if (!a || a === document.body) return 'body';
			const root = a.closest('[data-dse-element]');
			const ctl = a.closest('[data-dse-control]')?.getAttribute('data-dse-control');
			const what = ctl ?? (a.classList.contains('cm-content') ? 'cm-content' : a.tagName + '.' + String(a.className).slice(0, 30));
			const idx = root ? roots().indexOf(root) : -1;
			return what + '@root' + rid(root) + '/idx' + idx + (a.matches(':focus-visible') ? '+ring' : '');
		};
		roots().forEach(rid);
		const z = { t0, log: [], last: null, desc, roots, rid };
		z.timer = setInterval(() => { const d = desc(); if (d !== z.last) { z.log.push([T(), 'focus', d]); z.last = d; } }, 5);
		z.ref = app.vault.on('modify', (f) => z.log.push([T(), 'modify', f.path]));
		document.addEventListener('focusout', (e) => window.__zz602 === z && z.log.push([T(), 'out', e.target?.getAttribute?.('data-dse-control') ?? String(e.target?.className ?? e.target?.tagName).slice(0, 30)]), true);
		window.__zz602 = z; window.__dse602 = [];
		z.log.push([0, 'focus', desc()]); z.last = desc();
		return true;
	})()`);
const finish = async (probe: string): Promise<void> => {
	await dse.settle();
	await browser.pause(1000);
	const out = await dse.page(`(() => { const z = window.__zz602; clearInterval(z.timer); app.vault.offref(z.ref); return { final: z.desc(), roots: z.roots().map(z.rid), log: z.log, dbg: (window.__dse602 ?? []).map((e) => [Math.round(e[0] - z.t0), ...e.slice(1)]) }; })()`);
	rec(probe, 'result', out);
};
/** Programmatic keyboard-style focus (Tab in a CodeMirror editor would indent the note). */
