async function rMouse(t, at, { mods = 0, button = 'left', clickCount = 1, type = 'click' } = {}) {
	const buttons = button === 'right' ? 2 : 1;
	const base = { x: at.x, y: at.y, button, clickCount, modifiers: mods };
	if (type === 'click') {
		await t.cdp.call('Input.dispatchMouseEvent', { type: 'mousePressed', buttons, ...base });
		await t.cdp.call('Input.dispatchMouseEvent', { type: 'mouseReleased', buttons: 0, ...base });
	}
}
async function rAt(t, expr, noScroll = false) {
	const ok = await t.ev(`(() => { const el = ${expr}; if (!el) return null; if (!${noScroll}) el.scrollIntoView({ block: 'center', inline: 'center' }); const r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; })()`);
	t.expect(ok, `no visible element: ${expr}`);
	await sleep(250);
	await t.ev(`(${RG}.preClick = ${RG}.snap(), true)`);
	return t.ev(`(() => { const r = (${expr}).getBoundingClientRect(); return { x: r.left + r.width / 2, y: r.top + r.height / 2, l: r.left, r: r.right, t: r.top, b: r.bottom }; })()`);
}
async function rClick(t, expr, opts = {}) {
	const at = await rAt(t, expr, opts.noScroll);
	await t.cdp.call('Input.dispatchMouseEvent', { type: 'mouseMoved', x: at.x, y: at.y, modifiers: opts.mods ?? 0 });
	await rMouse(t, at, opts);
	return at;
}
async function rDbl(t, expr, n = 2) {
	const at = await rAt(t, expr);
	await t.cdp.call('Input.dispatchMouseEvent', { type: 'mouseMoved', x: at.x, y: at.y });
	for (let c = 1; c <= n; c++) await rMouse(t, at, { clickCount: c });
	return at;
}
async function rKey(t, key, code, vk, mods = 0, text) {
	await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyDown', key, code, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk, modifiers: mods, ...(text ? { text } : {}) });
	await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyUp', key, code, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk, modifiers: mods });
}
async function rType(t, s) {
	for (const ch of s) {
		await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyDown', key: ch, code: /\d/.test(ch) ? 'Digit' + ch : 'Key' + ch.toUpperCase(), text: ch, unmodifiedText: ch, windowsVirtualKeyCode: ch.toUpperCase().charCodeAt(0), nativeVirtualKeyCode: ch.toUpperCase().charCodeAt(0) });
		await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyUp', key: ch, code: /\d/.test(ch) ? 'Digit' + ch : 'Key' + ch.toUpperCase(), windowsVirtualKeyCode: ch.toUpperCase().charCodeAt(0), nativeVirtualKeyCode: ch.toUpperCase().charCodeAt(0) });
		await sleep(40);
	}
}
