#!/usr/bin/env python3
# usage: summarize.py base.jsonl [proto.jsonl]  -- tallies outcomes vs the oracle; with proto, the base -> proto flips.
import json, sys
from collections import Counter, defaultdict


def key(r):
    return (r['layout'], r['edit'], r['P'], r['link'], r['shape'], r['mode'], r['click'], r['plus'], r['timing'])


def load(p):
    d = {}
    for l in open(p):
        r = json.loads(l)
        d[key(r)] = r
    return d


def cls(o):
    return o.split(':')[0]


def judge(r):
    o = r.get('outcome', 'ERR')
    opts = r.get('opts')
    notes = []
    if 'err' in r:
        return ['ERROR']
    if o.startswith('CORRUPT'):
        notes.append('CORRUPT')
    if o == 'refused' and r['notices'] != ['disk']:
        notes.append('notice-on-refusal:' + ','.join(r['notices']))
    if o != 'refused' and r['notices']:
        notes.append('notice-on-save')
    if r['amb'] and r['twinned'] and o != 'refused':
        notes.append('WRITE-IN-AMBIGUOUS')
    if o.startswith('OTHER'):
        notes.append('OTHER-amb' if r['amb'] else ('OTHER-unamb-oracle-says-P' if opts and opts[0] == r['pA'] else 'OTHER-unamb-residual'))
    if o == 'refused' and not r['amb'] and opts and opts[0] == r['pA'] and r['pA'] is not None:
        notes.append('refused-unamb' + ('-headmap-agrees' if r['headMap'] == r['pA'] else ''))
    return notes


def tally(d, label):
    c = Counter(cls(r.get('outcome', 'ERR')) for r in d.values())
    j = Counter(n for r in d.values() for n in judge(r))
    print(f'== {label}: cells {len(d)}  outcomes {dict(c)}')
    print(f'   checks {dict(j)}')
    return c, j


def main():
    base = load(sys.argv[1])
    tally(base, sys.argv[1])
    show = int(sys.argv[3]) if len(sys.argv) > 3 else 12
    if len(sys.argv) > 2 and sys.argv[2] != '-':
        proto = load(sys.argv[2])
        tally(proto, sys.argv[2])
        flips = Counter()
        ex = defaultdict(list)
        for k, r in proto.items():
            b = base.get(k)
            if not b:
                flips[('missing-in-base', cls(r['outcome']))] += 1
                continue
            f = (cls(b['outcome']), cls(r['outcome']))
            flips[f] += 1
            if f[0] != f[1]:
                ex[f].append(r)
        print('   flips base->proto', {f'{a}->{b}': n for (a, b), n in sorted(flips.items())})
        for f, rs in ex.items():
            amb = Counter(('amb' if r['amb'] else 'unamb') + ('/twinned' if r['twinned'] else '') for r in rs)
            print(f'   {f[0]}->{f[1]}: {dict(amb)}')
            for r in rs[:show]:
                print(f"      {r['layout']} {r['edit']} {r['link']} {r['shape']}/{r['mode']} {r['click']} {r['plus']} {r['timing']} opts={r['opts']} pA={r['pA']} head={r['headMap']}")
    else:
        groups = defaultdict(list)
        for r in base.values():
            for n in judge(r):
                groups[n].append(r)
        for n, rs in groups.items():
            print(f'-- {n}: {len(rs)}')
            by = Counter((r['layout'], r['edit'], r['plus'], r['timing']) for r in rs)
            for (lay, ed, pl, tm), cnt in by.most_common(show):
                print(f'      {cnt:3d}  {lay} {ed} {pl} {tm}')


main()
