import sys,shutil
root, which = sys.argv[1], sys.argv[2]
orig = root + '/.orig'
files = {
 'twin': 'test/dom/framework/reading-mode-host-part-pin-twin.test.ts',
 'sweep': 'test/dom/framework/reading-mode-host-part-pin-sweep.test.ts',
 'host': 'src/framework/host/ReadingModeBlockHost.ts',
 'helper': 'test/dom/_settleVaultReads.ts',
}
import os
os.makedirs(orig, exist_ok=True)
for k,f in files.items():
    o = orig + '/' + k
    if not os.path.exists(o): shutil.copy(root + '/' + f, o)
    shutil.copy(o, root + '/' + f)
def rep(k, a, b):
    p = root + '/' + files[k]; s = open(p).read(); assert s.count(a) == 1, (k, a); open(p, 'w').write(s.replace(a, b))
SET = 'const settle = () => settleVaultReads();'
if which == 'S1-settle-noop':
    rep('twin', SET, 'const settle = () => Promise.resolve();'); rep('sweep', SET, 'const settle = () => Promise.resolve();')
elif which == 'S2-settle-one-hop':
    rep('twin', SET, 'const settle = () => new Promise((r) => setTimeout(r, 0));'); rep('sweep', SET, 'const settle = () => new Promise((r) => setTimeout(r, 0));')
elif which == 'H1-helper-counter-off':
    rep('helper', '\t\tinFlight++;\n', '\t\t// inFlight++ (mutant)\n'); rep('helper', '\t\t\tinFlight--;\n', '\t\t\t// inFlight-- (mutant)\n')
elif which == 'M0-fix-reverted':
    rep('host', 'if (this.twinned(whole, drawnPin) && !this.placedUniquely(whole, drawnPin, at.line)) return;', 'void drawnPin;')
elif which == 'M0+S1':
    rep('host', 'if (this.twinned(whole, drawnPin) && !this.placedUniquely(whole, drawnPin, at.line)) return;', 'void drawnPin;')
    rep('twin', SET, 'const settle = () => Promise.resolve();'); rep('sweep', SET, 'const settle = () => Promise.resolve();')
elif which == 'L1-no-cache-restore':
    rep('sweep', '\tcacheSpy.mockRestore();', '\tvoid cacheSpy;')
elif which == 'L2-restoreAllMocks':
    rep('sweep', '\tcacheSpy.mockRestore();', '\tjest.restoreAllMocks(); void cacheSpy;')
print('mutant', which)
