import sys
path, which = sys.argv[1], sys.argv[2]
src = open(path).read()
ORIG = "if (this.twinned(whole, drawnPin) && !this.placedUniquely(whole, drawnPin, at.line)) return;"
assert src.count(ORIG) == 1
M = {
    'orig': ORIG,
    'M0-no-gate': "void drawnPin;",
    'M2-drawn-only': "if (this.twinned(from.text, drawnPin) && !this.placedUniquely(whole, drawnPin, at.line)) return;",
    'M3-invert': "if (this.twinned(whole, drawnPin) && this.placedUniquely(whole, drawnPin, at.line)) return;",
    'M4-refuse-all-twinned': "if (this.twinned(whole, drawnPin)) return;",
    'M5-now-only': "if (this.twinned(whole, { text: whole, lineStart: 0 }) && !this.placedUniquely(whole, drawnPin, at.line)) return;",
    'M6-now-vs-now': "if (this.twinned(whole, drawnPin) && !this.placedUniquely(whole, { text: whole, lineStart: at.line }, at.line)) return;",
    'M7-no-twin-check': "if (!this.placedUniquely(whole, drawnPin, at.line)) return;",
}
open(path, 'w').write(src.replace(ORIG, M[which]))
print('mutated', which)
