async function r3Listeners(t, expr, type = 'click') {
	const r = await t.cdp.call('Runtime.evaluate', { expression: expr, objectGroup: 'r3' });
	const l = await t.cdp.call('DOMDebugger.getEventListeners', { objectId: r.result.objectId, objectGroup: 'r3' });
	const out = [];
	for (const x of l.listeners.filter((x) => x.type === type)) {
		let src = '';
		if (x.handler?.objectId) {
			const s = await t.cdp.call('Runtime.callFunctionOn', { objectId: x.handler.objectId, functionDeclaration: 'function () { return String(this).slice(0, 140); }', returnByValue: true });
			src = s.result?.value ?? '';
		}
		out.push({ capture: x.useCapture, scriptId: x.scriptId, line: x.lineNumber, col: x.columnNumber, src });
	}
	await t.cdp.call('Runtime.releaseObjectGroup', { objectGroup: 'r3' });
	return out;
}

async function r3OpenForm(t, K, mode, { plus = true, host = 'Lifecycle/gs83-host.md', src = 'Lifecycle/gs83-src.md', skipOpenHost = false } = {}) {
	if (!skipOpenHost) {
		await t.reset(src);
		await t.reset(host);
	}
	const link = skipOpenHost ? (mode === 'lp' ? `document.querySelector('.workspace-leaf.mod-active .cm-content .cm-hmd-internal-link')` : `document.querySelector('.workspace-leaf.mod-active .markdown-reading-view a.internal-link')`) : await K.openHost(host, mode);
	const hov = await K.hover(link, src);
	await K.moveTo(`${K.ENTRY(src)}.root`);
	await t.sleep(250);
	const pencil = `(${K.ENTRY(src)}.root.querySelector('.dse-chrome button[data-dse-chrome-item="edit"], button[aria-label^="Edit "]'))`;
	await t.waitFor(`!!${pencil}`, `${mode}: pencil`, 8000);
	if (plus) {
		await K.click(`${K.ENTRY(src)}.root.querySelector('button[aria-label^="Increase"]')`);
		await t.sleep(250);
	}
	await K.clickRevealed(pencil);
	await t.waitFor(`!!${K.FORM}`, `${mode}: form opened`, 10000);
	await t.sleep(400);
	return { hov, pencil };
}

async function r3OpenModal(t, K, mode, kind, { preClick = null } = {}) {
	const [src, host] = kind === 'stamina' ? ['Lifecycle/gs83-stam.md', 'Lifecycle/gs83-stam-host.md'] : ['Lifecycle/tracker.md', 'Lifecycle/gs83-trk-host.md'];
	const MODAL = kind !== 'cond' ? `document.querySelector('.modal-container .dse-modal')` : `document.querySelector('.modal-container .dse-condal-modal')`;
	await t.reset(src);
	await t.reset(host);
	const link = await K.openHost(host, mode);
	const hov = await K.hover(link, src);
	await K.moveTo(`${K.ENTRY(src)}.root`);
	await t.sleep(250);
	const opener = kind === 'stamina' ? `${K.ENTRY(src)}.root.querySelector('[data-dse-control="stamina.bar"]')` : `${K.ENTRY(src)}.root.querySelectorAll('.dse-cond--add')[0]`;
	await t.waitFor(`!!(${opener})`, `${mode} ${kind}: opener`, 8000);
	if (preClick) await preClick();
	await K.clickRevealed(opener);
	await t.waitFor(`!!${MODAL}`, `${mode} ${kind}: modal opened`, 8000);
	await t.sleep(400);
	return { hov, MODAL, src };
}
