import json, struct, sys, os
src, out = sys.argv[1], sys.argv[2]
with open(src,'rb') as f:
    data = f.read()
# asar: pickle uint32 size(4), header_size uint32; then pickle: uint32 payload, uint32 strlen, json
hsize = struct.unpack('<I', data[4:8])[0]
slen = struct.unpack('<I', data[12:16])[0]
header = json.loads(data[16:16+slen])
base = 8 + hsize
def walk(node, p):
    for name, n in node.get('files', {}).items():
        q = p + '/' + name
        if 'files' in n: walk(n, q)
        else:
            if 'offset' not in n: continue
            if q.endswith('.js') or q.endswith('.json') or q.endswith('.html'):
                off = base + int(n['offset']); sz = n['size']
                dst = out + q
                os.makedirs(os.path.dirname(dst), exist_ok=True)
                with open(dst,'wb') as g: g.write(data[off:off+sz])
walk(header, '')
