							await sleep(1500);
							await L('after real click on Save');
							rec.disk = t.read(SRC).includes('note: sc595a') ? 'saved' : 'not saved';
							rec.popoverAfterSave = await st();
							await mouse('mouseMoved', 700, 900);
							await sleep(1200);
							await L('pointer elsewhere 1.2 s after Save');
							await click(`document.querySelector('.workspace-leaf.mod-active .view-header-title, .workspace-leaf.mod-active .view-header')`);
							await sleep(300);
							await L('after a real click outside (view header)');
						} else if (variant === 'protoA-bg' || variant === 'protoA2-bg') {
							await mouse('mouseMoved', 30, 1000);
							await sleep(100);
							const bg = await t.ev(`(() => { const h = document.elementFromPoint(30, 1000); return h ? String(h.className) : null; })()`);
							await L('pointer on ' + bg);
							await mouse('mousePressed', 30, 1000, { button: 'left', clickCount: 1, buttons: 1 });
							await mouse('mouseReleased', 30, 1000, { button: 'left', clickCount: 1, buttons: 0 });
							await sleep(500);
							await L('after real click on the modal background');
							rec.disk = t.read(SRC).includes('note: sc595a') ? 'saved' : 'not saved';
						} else if (variant === 'protoA-cancel' || variant === 'protoA2-cancel') {
							await click(CANCEL);
							await sleep(500);
							await L('after real click on Cancel');
							rec.disk = t.read(SRC).includes('note: sc595a') ? 'saved' : 'not saved';
						} else {
							await t.key('Escape', 'Escape', 27);
							await sleep(500);
							await L('after Escape');
							rec.disk = t.read(SRC).includes('note: sc595a') ? 'saved' : 'not saved';
						}
					} else if (variant === 'protoB') {
						await t.ev(`(() => { const e = ${ENTRY(SRC)}; const reg = app.plugins.plugins['draw-steel-elements'].viewRegistry; e.host.keepFormOpenOn = () => reg; return true; })()`);
						await L('protoB installed (host.keepFormOpenOn -> registry)');
						await click(RAW);
						await sleep(500);
						await L('after real click in raw box');
						const alive = await t.ev(`!!${FORM}`);
						if (alive) {
							await t.ev(`(() => { const ta = ${RAW}; ta.focus(); ta.setSelectionRange(ta.value.length, ta.value.length); return true; })()`);
							await cdp.call('Input.insertText', { text: '\nnote: sc595b' });
							await sleep(200);
							await click(RAW);
							await sleep(300);
							await L('after second real click in raw box (typed)');
							await click(SAVE);
							await sleep(1500);
							await L('after real click on Save');
						}
						rec.disk = t.read(SRC).includes('note: sc595b') ? 'saved' : 'not saved';
						rec.notices = await t.ev('__lc.notices.slice(-3).map((x) => x.text)');
					}
					Object.assign(rec, await dump());
				} catch (e) {
					rec.error = e.message;
					try {
						Object.assign(rec, await dump());
					} catch {
						/* */
					}
				} finally {
					await closeAll();
				}
				emit(rec);
			};

			/** Q5: another DSE modal opened from a hover preview card. */
			const otherModal = async (mode, kind) => {
				const label = `${mode}/other-${kind}`;
				const rec = { label };
				const kbd = kind === 'stamina-kbd';
				if (kbd) kind = 'stamina';
				const src = kind === 'stamina' ? 'Lifecycle/sc595-stam.md' : 'Lifecycle/tracker.md';
				const host = kind === 'stamina' ? 'Lifecycle/sc595-stam-host.md' : 'Lifecycle/sc595-trk-host.md';
				try {
					await t.reset(src);
					await t.reset(host);
					const link = await openHost(host, mode);
					rec.hover = await hover(link, src);
					rec.install = await t.ev(INSTALL);
					await moveTo(`${ENTRY(src)}.root`);
					await sleep(250);
					const opener = kind === 'stamina' ? `${ENTRY(src)}.root.querySelector('[data-dse-control="stamina.bar"]')` : `${ENTRY(src)}.root.querySelectorAll('.dse-cond--add')[0]`;
					await t.waitFor(`!!(${opener})`, `${label}: opener`, 8000);
					if (kbd) {
						await t.ev(`(${opener}).focus()`);
						await L('bar focused by script (no click anywhere)');
						await t.key('Enter', 'Enter', 13);
					} else await clickRevealed(opener);
					const MSEL = kind === 'stamina' ? `document.querySelector('.modal-container .dse-modal')` : `document.querySelector('.modal-container .dse-condal-modal')`;
					await t.waitFor(`!!(${MSEL})`, `${label}: modal`, 8000);
					await sleep(300);
					await L('modal open');
					rec.modalFocusedBefore = await st();
					if (kbd) {
						await sleep(1500);
						await L('pointer left where it was, 1.5 s');
						await moveTo(`(${MSEL}).closest('.modal').querySelector('.modal-title') || (${MSEL})`);
						await sleep(1500);
						await L('pointer rested 1.5 s on the modal title (no click)');
						rec.modalAfterRest = await t.ev(`!!(${MSEL})`);
						if (!rec.modalAfterRest) throw new Error('the modal closed with no click');
					}
					// click a NON-acting spot inside the modal: its title
					await click(`(${MSEL}).closest('.modal').querySelector('.modal-title') || (${MSEL})`);
					await sleep(400);
					await L('after real click on the modal title');
					rec.modalAfter = await t.ev(`!!(${MSEL})`);
					Object.assign(rec, await dump());
				} catch (e) {
					rec.error = e.message;
					try {
						Object.assign(rec, await dump());
					} catch {
						/* */
					}
				} finally {
					await closeAll();
				}
				emit(rec);
			};

			await t.ev(`app.plugins.plugins['draw-steel-elements'].frameworkV2.services.prefs.set('authoringControls', true)`);
			await sleep(500);
			const which = (process.env.SC595_TRIALS ?? 'rv/press-hold,lp/press-hold').split(',');
			for (const w of which) {
				const [mode, variant] = w.split('/');
				if (variant.startsWith('other-')) await otherModal(mode, variant.slice(6));
				else await trial(mode, variant);
			}
			return `${out.length} trials`;
		},
	};
}
