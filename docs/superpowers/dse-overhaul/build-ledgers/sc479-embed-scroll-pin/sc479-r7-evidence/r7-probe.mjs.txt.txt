/** park: 'inside' = viewport top `off` px under the tracker; 'trk' = viewport top `off` px under the tracker's top (tracker on screen). */
async function r7park(t, mode, off) {
	let geo = null;
	for (let i = 0; i < 3; i++) {
		geo = await t.ev(`(() => { const s = ${r3S('window.__r3A')}; const e = window.__r3emb(s); if (!e) return null; const trk = e.querySelector('[data-dse-element="initiative"]'); const sr = s.getBoundingClientRect(); const tr = trk.getBoundingClientRect(); s.scrollTop = (${JSON.stringify(mode)} === 'trk' ? tr.top : tr.bottom) - sr.top + s.scrollTop + ${off}; const er = e.getBoundingClientRect(); const sz = s.querySelector(':scope > .markdown-preview-sizer'); return { st: Math.round(s.scrollTop), maxSt: s.scrollHeight - s.clientHeight, sh: s.scrollHeight, sizerInlineMH: sz ? parseFloat(sz.style.minHeight) : null, sizerH: sz ? Math.round(sz.getBoundingClientRect().height) : null, embTop: Math.round(er.top - sr.top), embBottom: Math.round(er.bottom - sr.top), trkTop: Math.round(trk.getBoundingClientRect().top - sr.top), trkBottom: Math.round(trk.getBoundingClientRect().bottom - sr.top) }; })()`);
		await t.sleep(450);
	}
	return geo;
}
