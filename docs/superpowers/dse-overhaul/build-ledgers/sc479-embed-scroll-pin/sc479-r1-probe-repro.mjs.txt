/** Scroll leaf L (Reading or LP) until A's embed is drawn, then park it. Returns the parked position. */
async function p479Park(t, L, where) {
	const sc = `(window.__p479${L}.view.getMode() === 'preview' ? window.__p479.outerOf(window.__p479${L}) : window.__p479.lpOf(window.__p479${L}))`;

	await t.sleep(400);
	let found = false;
	for (let i = 0; i < 40 && !found; i++) {
		found = await t.ev(`(() => { const s = ${sc}; const e = Array.from(s.querySelectorAll('.internal-embed')).find((x) => !x.parentElement.closest('.internal-embed')); if (e) { e.scrollIntoView({ block: 'start' }); return true; } s.scrollTop += 400; return false; })()`);
		await t.sleep(350);
	}
	for (let i = 0; i < 30; i++) {
		if (await t.ev(`(() => { const s = ${sc}; const e = window.__p479.embOf(s); return !!(e && e.querySelector('[data-dse-element]')); })()`)) break;
		await t.sleep(200);
	}
	await t.sleep(600);
	const pos = `(() => { const s = ${sc}; const e = window.__p479.embOf(s); const sr = s.getBoundingClientRect(); const max = s.scrollHeight - s.clientHeight; if (!e) return { err: 'no embed' }; const er = e.getBoundingClientRect(); const top = er.top - sr.top + s.scrollTop; return { top: Math.round(top), h: Math.round(er.height), max, ch: s.clientHeight }; })()`;
	for (let i = 0; i < 5; i++) {
		const p = await t.ev(pos);
		if (p.err) {
			if (where === 'end') await t.ev(`(() => { const s = ${sc}; s.scrollTop = s.scrollHeight; return true; })()`);
			await t.sleep(400);
			continue;
		}
		const target = where === 'above' ? Math.max(200, p.top - p.ch - 600) : where === 'in' ? p.top + 1500 : where === 'below' ? p.top + p.h + 400 : p.max + 100000;
		await t.ev(`(() => { const s = ${sc}; s.scrollTop = ${target}; return true; })()`);
		await t.sleep(450);
	}
	await t.sleep(400);
	const end = await t.ev(`(() => { const s = ${sc}; const e = window.__p479.embOf(s); const sr = s.getBoundingClientRect(); return { st: Math.round(s.scrollTop), sh: s.scrollHeight, ch: s.clientHeight, max: s.scrollHeight - s.clientHeight, embTop: e ? Math.round(e.getBoundingClientRect().top - sr.top) : null, embH: e ? Math.round(e.getBoundingClientRect().height) : null }; })()`);
	return Object.assign({ embStart: await t.ev(pos) }, end);
}
