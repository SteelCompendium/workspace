async function r3open(t, rel, leafVar = '__r3A', needDse = true) {

async function r3seek(t, leaf) {
	await t.ev(`(() => { ${r3S(leaf)}.scrollTop = 0; return true; })()`);
	await t.sleep(300);
	for (let i = 0; i < 60; i++) {
		if (await t.ev(`!!window.__r3emb(${r3S(leaf)})`)) return;
		await t.ev(`(() => { const s = ${r3S(leaf)}; s.scrollTop += Math.round(s.clientHeight * 0.6); return true; })()`);
		await t.sleep(200);
	}
	throw new Error('the embed never rendered while seeking');
}
async function r3park(t, leaf, how) {
	if (typeof how === 'string' && how !== 'end') await r3seek(t, leaf);
	for (let i = 0; i < 3; i++) {
