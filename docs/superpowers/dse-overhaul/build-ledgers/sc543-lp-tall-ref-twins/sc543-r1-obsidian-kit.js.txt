/** Page-side instrumentation: lookups (refs.resolve), Obsidian's LP code-block widget lifecycle (initDOM/toDOM/destroy), a probe plugin. */
const S43_PAGE = `(() => {

	S.now = () => Math.round(performance.now() - S.t0);
	if (!S.vaultHooked) {
		S.vaultHooked = true;
		for (const fn of ['cachedRead', 'read']) {
			const r = app.vault[fn].bind(app.vault);
			app.vault[fn] = (f) => { S.pendingReads++; return r(f).finally(() => S.pendingReads--); };
		}
	}
	const refs = app.plugins.plugins['draw-steel-elements'].frameworkV2.services.refs;
	if (!refs.__s43) {
		refs.__s43 = true;
		const real = refs.resolve.bind(refs);
		refs.resolve = (raw, sp) => { S.calls.push({ raw, sp, t: S.now() }); S.pendingResolve++; return real(raw, sp).finally(() => { S.pendingResolve--; }); };
	}
