import sys
name=sys.argv[1]; p=sys.argv[2]
s=open(p,encoding='utf-8').read()
def sub(old,new):
    global s
    assert s.count(old)==1,(name,old,s.count(old))
    s=s.replace(old,new)

    sub("if (parent && targets.some((t) => parent.targets.has(t))) {","if (parent && false) {")
elif name=='M2-drop-cap':

elif name=='M3-drop-delete-at-onMount':
    sub("\t\tREF_CHAIN.delete(root);\n","")
elif name=='M4-claim-before-last-gen-check':
    sub("""		const rawText = await this.legacyRefRawText(resolved);
		if (gen !== this.resolveGen) return;
		const model = this.base.parse(resolved.data, rawText);
		// SC-536: the last gen check is above (see resolveAndMount's SCC ladder).
		if (!this.claimChain(root, raw, [resolved.file ? `file:${resolved.file.path}` : `raw:${this.base.id}|${raw}`])) return;
""","""		const rawText = await this.legacyRefRawText(resolved);
		if (!this.claimChain(root, raw, [resolved.file ? `file:${resolved.file.path}` : `raw:${this.base.id}|${raw}`])) return;
		if (gen !== this.resolveGen) return;
		const model = this.base.parse(resolved.data, rawText);
""")

    sub("[resolved.file ? `file:${resolved.file.path}` : `raw:${this.base.id}|${raw}`]","[`raw:${this.base.id}|${raw}`]")

    sub("for (let n = root.parentElement; n; n = n.parentElement) {","for (let n = root.parentElement; n; n = null) {")
elif name=='M7-scc-drop-file-key':
    sub("[`scc:${code}`, `file:${entity.file.path}`]","[`scc:${code}`]")
elif name=='M8-scc-claim-before-last-gen-check':
    sub("""		const body = await entity.body();
		if (gen !== this.resolveGen) return;
		const source: RefSource = { file: entity.file, frontmatter: entity.frontmatter, body };
		// SC-536: the last gen check is above; from here this resolution is the one that draws, so it may record its chain.

    sub("""		const rawText = await this.legacyRefRawText(resolved);
		if (gen !== this.resolveGen) return;
""","""		const rawText = await this.legacyRefRawText(resolved);

    raise SystemExit('unknown '+name)
