function sc536Kit(t) {
	const K = {
		tag: () => {
			const b = process.env.DSE_LIFECYCLE_BUNDLE ?? '';
			return b.includes('bundle-base') ? 'base' : b.includes('bundle-instr') ? 'instr' : b.includes('bundle-proto') ? 'proto' : 'repo';
		},
		/** Hooks: services.refs.resolve on the CURRENT plugin instance (re-run after a plugin re-enable) and app.vault.read/cachedRead (survives a re-enable). */
		async install() {
			await t.ev(`(() => {
				const K = (window.__k536 ??= { calls: [], pendingResolve: 0, pendingReads: 0, holdRefs: false, holdPath: null, held: [], reads: [] });
				if (!K.vaultHooked) {
					K.vaultHooked = true;
					for (const fn of ['cachedRead', 'read']) {
						const r = app.vault[fn].bind(app.vault);
						app.vault[fn] = (f) => {
							K.pendingReads++;
							K.reads.push(f && f.path);
							const go = () => r(f).finally(() => K.pendingReads--);
							if (K.holdPath && f && f.path === K.holdPath) return new Promise((res, rej) => K.held.push({ kind: 'read', path: f.path, go: () => go().then(res, rej) }));
							return go();
						};
					}
				}
				const refs = app.plugins.plugins['draw-steel-elements'].frameworkV2.services.refs;
				if (!refs.__k536) {
					refs.__k536 = true;
					const real = refs.resolve.bind(refs);
					refs.resolve = (raw, sp) => {
						const rec = { raw, sp };
						K.calls.push(rec);
						K.pendingResolve++;
						const go = () => real(raw, sp).finally(() => { K.pendingResolve--; });
						if (K.holdRefs) return new Promise((res, rej) => K.held.push({ kind: 'resolve', raw, go: () => go().then(res, rej) }));
						return go();
					};
				}
				return true;
			})()`);
		},
		holdRefs: (on) => t.ev(`(() => { window.__k536.holdRefs = ${on ? 'true' : 'false'}; return true; })()`),
		holdPath: (p) => t.ev(`(() => { window.__k536.holdPath = ${JSON.stringify(p)}; return true; })()`),
		held: () => t.ev(`window.__k536.held.length`),
		calls: () => t.ev(`window.__k536.calls.length`),
		/** Release one held call (FIFO or LIFO); returns how many were held before. */
		releaseOne: (lifo) => t.ev(`(() => { const H = window.__k536.held; if (!H.length) return 0; const n = H.length; (${lifo ? 'H.pop()' : 'H.shift()'}).go(); return n; })()`),
		async releaseAll(lifo) {
			for (;;) {
				const n = await K.releaseOne(lifo);
				if (!n) return;
				await K.settle();
			}
		},
		/** Settled = every lookup / read in flight is one we hold, then two painted frames, then the same again (a condition, not a delay). */
		async settle(ms = 15000) {
			const C = `(() => { const K = window.__k536; const hr = K.held.filter((h) => h.kind === 'read').length; const hs = K.held.filter((h) => h.kind === 'resolve').length; return K.pendingReads === hr && (hr > 0 || K.pendingResolve === hs); })()`;
			await t.waitFor(C, 'lookups/reads settled', ms);
			await t.ev(`new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(() => r(true))))`);
			await t.waitFor(C, 'lookups/reads settled (2)', ms);
		},
		/** Top-level feature root of a leaf's reading view. */
		ROOT: (leaf) => `(Array.from(${t.readingScope(leaf)}.querySelectorAll('[data-dse-element="feature"]')).filter((x) => x.isConnected && !x.parentElement.closest('[data-dse-element]'))[0])`,
		async st(leaf, card = 'Ref Strike') {
			return t.ev(`(() => { const r = ${K.ROOT(leaf)}; const tx = r ? r.textContent : null; return { present: !!r, card: !!tx && /${card}/.test(tx), nesting: !!tx && /nesting too deep/.test(tx), cards: r ? (tx.match(/${card}/g) || []).length : 0, text: tx ? tx.slice(0, 100) : null }; })()`);
		},
		/** Whole-tree stats of every feature element under a scope expression. */
		treeStats: (scopeExpr, card) => t.ev(`(() => { const sc = ${scopeExpr}; if (!sc) return null; const all = Array.from(sc.querySelectorAll('[data-dse-element="feature"]')); const depth = (el) => { let d = 0; for (let p = el.parentElement; p; p = p.parentElement) if (p.matches && p.matches('[data-dse-element]')) d++; return d; }; return { roots: all.length, maxDepth: all.reduce((a, el) => Math.max(a, depth(el)), 0), errorCards: Array.from(sc.querySelectorAll('.dse-error-card')).map((e) => e.textContent.slice(0, 140)), cards: (sc.textContent.match(new RegExp(${JSON.stringify(card)}, 'g')) || []).length, lookups: window.__k536.calls.length }; })()`),

}
