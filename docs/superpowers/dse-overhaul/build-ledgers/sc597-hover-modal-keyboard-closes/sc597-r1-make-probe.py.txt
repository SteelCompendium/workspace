# Builds the SC-597 scratch probe harness: a copy of the worktree's visual-harness/obsidian-lifecycle.mjs (dse 94133a5a) with the
# SC597-PROBE scenario (sc597-r1-probe-scenario.js) injected before `const SCENARIOS = [` and put first in SCENARIOS.
# Nothing in the worktree is changed. The copy lives in <scratch>/harness/visual-harness/ next to a <scratch>/harness/scripts symlink.
# usage: python3 -I sc597-r1-make-probe.py <worktree dse dir> <scenario.js> <out .mjs>
import sys
W, scen, dst = sys.argv[1:]
s = open(W + '/visual-harness/obsidian-lifecycle.mjs').read()
inj = open(scen).read()
def rep(a, b):
    global s
    assert s.count(a) == 1, (a, s.count(a))
    s = s.replace(a, b)
rep("const repo = path.dirname(dir);", f"const repo = '{W}';")
rep("const SCENARIOS = [", inj + "\nconst SCENARIOS = [\n\tSC597_SCENARIO,")
open(dst, 'w').write(s)
print('wrote', dst)
