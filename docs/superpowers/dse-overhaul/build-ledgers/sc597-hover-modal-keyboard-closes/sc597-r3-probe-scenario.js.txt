async function r3Key(t, which, withText) {
	const k = which === 'space' ? [' ', 'Space', 32, ' '] : ['Enter', 'Enter', 13, '\r'];
	const down = { type: 'keyDown', key: k[0], code: k[1], windowsVirtualKeyCode: k[2], nativeVirtualKeyCode: k[2] };
	if (withText) Object.assign(down, { text: k[3], unmodifiedText: k[3] });
	await t.cdp.call('Input.dispatchKeyEvent', down);
	await t.cdp.call('Input.dispatchKeyEvent', { type: 'keyUp', key: k[0], code: k[1], windowsVirtualKeyCode: k[2], nativeVirtualKeyCode: k[2] });
}

async function r3Trial(t, trial) {
	const K = gs83Kit(t);
	const rec = { trial };
	try {
		if (trial.startsWith('pv-')) {
			const isMont = trial.startsWith('pv-mont');
			const [src, host] = isMont ? ['Lifecycle/r3-mont.md', 'Lifecycle/r3-mont-host.md'] : ['Lifecycle/gs83-stam.md', 'Lifecycle/gs83-stam-host.md'];
			const OPENER = isMont ? `(${K.ENTRY(src)}).root.querySelector('.dse-mt__cell[role="button"][tabindex="0"]')` : `(${K.ENTRY(src)}).root.querySelector('[data-dse-control="stamina.bar"]')`;
			await t.reset(src);
			await t.reset(host);
			const link = await K.openHost(host, 'rv');
			rec.hover = await K.hover(link, src);
			await t.waitFor(`!!(${OPENER})`, `${trial}: opener drawn`, 8000);
			await K.moveTo(`${K.ENTRY(src)}.root`);
			await t.sleep(400);
			await t.ev(`(() => { (${OPENER}).focus(); return true; })()`);
			await t.ev(R3_LOG);
			const which = trial.endsWith('space') ? 'space' : 'enter';
			await r3Key(t, which, true);
			await t.ev(`(() => { window.__r3arm = false; return true; })()`);
			const escAt = trial === 'pv-esc40' ? 40 : trial === 'pv-esc200' ? 200 : null;
			if (escAt !== null) {
				await t.sleep(escAt);
				rec.beforeEsc = await t.ev(R3_STATE);
				await t.key('Escape', 'Escape', 27);
				await t.sleep(600);
				rec.afterEsc = await t.ev(R3_STATE);
				await t.sleep(900);
				rec.at1500 = await t.ev(R3_STATE);
			} else {
				await t.sleep(150);
				rec.at150 = await t.ev(R3_STATE);
				await t.sleep(1350);
				rec.at1500 = await t.ev(R3_STATE);
				await t.key('Escape', 'Escape', 27);
				await t.sleep(500);
				rec.afterEsc = await t.ev(R3_STATE);
			}
			rec.log = await t.ev(`window.__r3`);
			rec.note = t.read(src);
		} else if (trial === 'lp-callout') {
			const rel = 'Lifecycle/callout-stamina-lp.md';
			const LP = LC_MODES[1];
			await enterMode(t, rel, LP, '[data-dse-element="stamina-bar"]');
			const PARKED = FIXTURES[rel].split('\n').length - 2;
			await t.ev(`app.workspace.getMostRecentLeaf().view.editor.setCursor({ line: ${PARKED}, ch: 0 })`);
			await t.sleep(400);
			const root = `Array.from(app.workspace.getMostRecentLeaf().view.containerEl.querySelectorAll('.markdown-source-view [data-dse-element="stamina-bar"]')).find((r) => r.isConnected)`;
			const st = () => t.ev(`(() => { const r = ${root}; const ed = app.workspace.getMostRecentLeaf().view.editor; const sel = ed.listSelections()[0]; return { cursor: ed.getCursor().line, sel: [sel.anchor.line, sel.anchor.ch, sel.head.line, sel.head.ch], widget: !!(r && r.closest('.cm-callout')), interactiveChild: !!(r && r.classList.contains('interactive-child')), modals: document.querySelectorAll('.modal-container').length, scroll: Math.round(app.workspace.getMostRecentLeaf().view.containerEl.querySelector('.cm-scroller').scrollTop) }; })()`);
			rec.before = await st();
			await t.ev(`(() => { (${root}).querySelector('[data-dse-control="stamina.bar"]').focus(); return true; })()`);
			await t.ev(R3_LOG);
			await r3Key(t, 'enter', true);
			await t.ev(`(() => { window.__r3arm = false; return true; })()`);
			await t.sleep(800);
			rec.afterKey = await st();
			await t.key('Escape', 'Escape', 27);
			await t.sleep(500);
			rec.afterEsc = await st();
			rec.log = await t.ev(`window.__r3`);
		} else if (trial === 'lp-embed') {
			const rel = 'Lifecycle/r3-embhost.md';
			await t.reset('Lifecycle/gs83-stam.md');
			await t.open(rel);
			await t.reset(rel);
			await t.ev(`(async () => { await app.workspace.getMostRecentLeaf().setViewState({ type: 'markdown', state: { file: ${JSON.stringify(rel)}, mode: 'source', source: false }, active: true }); })()`);
			await t.sleep(1500);
			await t.ev(`app.workspace.getMostRecentLeaf().view.editor.setCursor({ line: 0, ch: 0 })`);
			const root = `Array.from(app.workspace.getMostRecentLeaf().view.containerEl.querySelectorAll('.markdown-source-view .internal-embed [data-dse-element="stamina-bar"]')).find((r) => r.isConnected)`;
			// scroll the embed into the middle of the pane (the editor draws it lazily)
			for (let i = 0; i < 20; i++) {
				const ok = await t.ev(`(() => { const r = ${root}; if (r) { r.scrollIntoView({ block: 'center' }); return true; } const sc = app.workspace.getMostRecentLeaf().view.containerEl.querySelector('.cm-scroller'); sc.scrollTop += 400; return false; })()`);
				await t.sleep(250);
				if (ok) break;
			}
			await t.waitFor(`!!(${root})`, 'lp-embed: the embedded stamina bar is drawn', 8000);
			await t.ev(`(() => { (${root}).scrollIntoView({ block: 'center' }); return true; })()`);
			await t.sleep(600);
			const st = () => t.ev(`(() => { const r = ${root}; const ed = app.workspace.getMostRecentLeaf().view.editor; const sel = ed.listSelections()[0]; return { cursor: ed.getCursor().line, sel: [sel.anchor.line, sel.anchor.ch, sel.head.line, sel.head.ch], drawn: !!r, interactiveChild: !!(r && r.classList.contains('interactive-child')), modals: document.querySelectorAll('.modal-container').length, scroll: Math.round(app.workspace.getMostRecentLeaf().view.containerEl.querySelector('.cm-scroller').scrollTop) }; })()`);
			rec.before = await st();
			await t.ev(`(() => { (${root}).querySelector('[data-dse-control="stamina.bar"]').focus({ preventScroll: true }); return true; })()`);
			rec.afterFocus = await st();
			await t.ev(R3_LOG);
			await r3Key(t, 'enter', true);
			await t.ev(`(() => { window.__r3arm = false; return true; })()`);
			await t.sleep(800);
			rec.afterKey = await st();
			await t.key('Escape', 'Escape', 27);
			await t.sleep(500);
			rec.afterEsc = await st();
			rec.log = await t.ev(`window.__r3`);
		}
	} catch (e) {
		rec.error = e.message;
	} finally {
		await K.cleanup();
	}
	return rec;
}
