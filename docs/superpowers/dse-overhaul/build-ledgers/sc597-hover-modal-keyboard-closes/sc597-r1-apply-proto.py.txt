# SC-597 r1 prototype: one bundle, three candidate fixes behind a runtime switch window.__sc597 ('a' | 'b1' | 'b1r' | 'c'); unset = base behaviour.
#   (a)   per-opener synthetic click: Enter/Space on the stamina bar / a montage cell calls el.click() instead of the open callback
#   (b1)  DseModal: opened from a preview outside a click -> find the HoverPopover instance, setIsFocused(true) (never reset)
#   (b1r) as b1, and setIsFocused(false) again when the modal closes
#   (c)   DseModal: opened from a preview outside a click -> dispatch a synthetic click on the anchor AFTER focus moved into the modal,
#         arm the 10/150 ms re-pushes from the window-bubble phase of that click (same ordering trick as a real opening click)
# usage: python3 -I sc597-r1-apply-proto.py <dse copy root>
import sys, os
root = sys.argv[1]

def patch(rel, pairs):
    p = os.path.join(root, rel)
    s = open(p).read()
    for a, b in pairs:
        assert s.count(a) == 1, (rel, a, s.count(a))
        s = s.replace(a, b)
    open(p, 'w').write(s)
