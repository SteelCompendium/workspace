/** The note: an identical construct (the twin, B) above the `## Tracker` section that holds A. Block shapes carry ids (`^a` names A). */
function noteOf(build: (v: number) => string[], plain: boolean, a: number, b: number, added: boolean, ids: boolean): { lines: string[]; aStart: number; bStart: number; cStart: number } {
	const idLines = (id: string) => (!ids ? [] : plain ? [`^${id}`] : ['', `^${id}`]);
	const lines = ['# N', ''];
	let cStart = -1;
	if (added) {
		cStart = lines.length;
		lines.push(...build(a === 9 ? 5 : a), '', 'p-added', '');
	}
	const bStart = lines.length;
	lines.push(...build(b), ...idLines('b'), '', 'p-twin', '', '## Tracker', '');
	const aStart = lines.length;
	lines.push(...build(a), ...idLines('a'), '', '## After', '', 'END');
	return { lines, aStart, bStart, cStart };
}
const pos = (line: number) => ({ line, col: 0, offset: 0 });
function cacheFor(lines: string[], aStart: number, aEnd: number, bStart: number, bEnd: number) {
	const headings: Array<{ heading: string; level: number; position: unknown }> = [];
	lines.forEach((l, i) => {
		const h = /^(#{1,6})\s+(.*)$/.exec(l);
		if (h) headings.push({ heading: h[2], level: h[1].length, position: { start: pos(i), end: pos(i) } });
	});
	return { headings, blocks: { a: { id: 'a', position: { start: pos(aStart), end: pos(aEnd) } }, b: { id: 'b', position: { start: pos(bStart), end: pos(bEnd) } } } };
}
function popoverEl(): { popover: HTMLElement; el: HTMLElement } {
	const popover = document.createElement('div');
	popover.className = 'popover hover-popover';
	const embed = popover.appendChild(document.createElement('div'));
	embed.className = 'markdown-embed';
	const el = embed.appendChild(document.createElement('div'));
	document.body.appendChild(popover);
	return { popover, el };
}
function embedEl(src: string, lp: boolean): HTMLElement {
	const embed = document.createElement('div');
	embed.className = 'internal-embed markdown-embed';
	embed.setAttribute('src', src);
	const el = embed.appendChild(document.createElement('div'));
	if (lp) document.createElement('div').appendChild(embed).parentElement!.className = 'cm-embed-block';
	return el;
}
const settle = () => new Promise((r) => setTimeout(r, 25));

/** One cell. Returns 'A' (the clicked block), 'TWIN', 'ADDED', 'refused+Notice', 'refused (no Notice)'. */
async function cell(cname: string, build: (v: number) => string[], off: number, plain: boolean, shape: Shape, hostKind: 'rv' | 'lp', edit: Edit, link = true): Promise<string> {
	resetDroppedWriteNotices();
	Notice.notices.length = 0;
	resetLinks();
	const ids = shape.startsWith('block');
	const pre = noteOf(build, plain, 5, 5, false, ids);
	const len = build(5).length;
	const preText = pre.lines.join('\n');
	const now = noteOf(build, plain, edit === 'own edited' ? 9 : 5, edit === 'twin edited' ? 9 : 5, edit === 'identical added', ids);
	const nowText = now.lines.join('\n');
	const app = new App();
	app.vault.setFile('Note.md', nowText);
	jest.spyOn(app.metadataCache, 'getFileCache').mockImplementation(() => cacheFor(now.lines, now.aStart, now.aStart + len - 1, now.bStart, now.bStart + len - 1) as never);
	// the part Obsidian drew BEFORE the outside edit (the preview / embed is not redrawn yet)
	const aLines = build(5);
	let sec: Sec;
	let el: HTMLElement;
	let linktext: string | null = null;
	const winStart = plain ? off : 0;
	const winEnd = plain ? off + 3 : len - 1;
	if (shape === 'heading hover') {
		const part = ['', ...aLines, '', ''];
		sec = { text: part.join('\n'), lineStart: 1 + winStart, lineEnd: 1 + winEnd };
		linktext = 'Note#Tracker';
	} else if (shape === 'block hover') {
		sec = { text: aLines.join('\n'), lineStart: winStart, lineEnd: winEnd };
		linktext = 'Note#^a';
	} else if (shape === 'heading embed') {
		const part = ['## Tracker', '', ...aLines, ''];
		sec = { text: part.join('\n'), lineStart: 2 + winStart, lineEnd: 2 + winEnd };
	} else if (shape === 'block embed') {
		sec = { text: aLines.join('\n'), lineStart: winStart, lineEnd: winEnd };
	} else {
		sec = { text: preText, lineStart: pre.aStart + winStart, lineEnd: pre.aStart + winEnd };
		linktext = 'Note';
	}
	let popover: HTMLElement | null = null;
	if (shape.endsWith('hover')) {
		const p = popoverEl();
		popover = p.popover;
		el = p.el;
	} else el = embedEl(shape === 'heading embed' ? 'Note#Tracker' : 'Note#^a', hostKind === 'lp');
	if (link && linktext && popover) noteLink({ linktext, sourcePath: 'Host.md', hoverParent: { hoverPopover: { hoverEl: popover } } });
	const section: Section = { current: sec };
	const host = new ReadingModeBlockHost(new Plugin(app) as never, el, switchableCtx(section) as never, 'ds-counter');
	host.setMountedBody(bodyOf(5));
	host.notePersistIntent();
	await settle();
	const wrote = await host.replaceSource(bodyOf(6), { fromView: true });
	popover?.remove();
	const after = app.vault.getContent('Note.md')!;
	if (after === nowText) return Notice.notices.length ? 'refused+Notice' : 'refused (no Notice)';
	const al = after.split('\n');
	const valueAt = (start: number) => {
		for (let i = start; i < start + len; i++) {
			const m = /current_value: (\d+)/.exec(al[i] ?? '');
			if (m) return Number(m[1]);
		}
		return -1;
	};
	const got: string[] = [];
	if (valueAt(now.aStart) === 6) got.push('A');
	if (valueAt(now.bStart) === 6) got.push('TWIN');
	if (now.cStart >= 0 && valueAt(now.cStart) === 6) got.push('ADDED');
	return `${got.join('+') || 'OTHER'}${wrote ? '' : '(false)'}${Notice.notices.length ? '+Notice' : ''}`;
}

describe('SC-594 r1 sweep', () => {
	let warn: jest.SpyInstance;
	beforeEach(() => {
		warn = jest.spyOn(console, 'warn').mockImplementation(() => {});
	});
	afterEach(() => warn.mockRestore());
	const cases = CONSTRUCTS.flatMap(([cname, build, off, plain]) =>
		SHAPES.flatMap((shape) => (['rv', 'lp'] as const).flatMap((hostKind) => EDITS.map((edit) => ({ cname, build, off, plain, shape, hostKind, edit })))),
	);
	test.each(cases)('$cname | $shape | $hostKind | $edit', async ({ cname, build, off, plain, shape, hostKind, edit }) => {
		const out = await cell(cname, build, off, plain, shape, hostKind, edit);
		record(`${cname}|${shape}|${hostKind}|${edit}|${out}`);
		expect(out).not.toMatch(/TWIN|ADDED/); // the invariant (recorded either way)
	});
	// the hover-link record absent (a popover the page preview did not open): what the rule falls back to
	test.each(CONSTRUCTS.slice(0, 2).flatMap(([cname, build, off, plain]) => (['heading hover', 'block hover'] as const).map((shape) => ({ cname, build, off, plain, shape }))))(
		'no hover-link record | $cname | $shape',
		async ({ cname, build, off, plain, shape }) => {
			const out = await cell(cname, build, off, plain, shape, 'rv', 'none', false);
			record(`${cname}|${shape}|rv|none, NO hover-link record|${out}`);
		},
	);
});

		rec('T14 immediate flush', `${r} ${JSON.stringify(vals(app))} notices ${Notice.notices.length}`);
		expect(vals(app)).toEqual([6, 5]);
	});
	test('T15 [[Note#^a]] hover, the cache lags and points ^a at the twin B (which carries ^b): never B', async () => {
		const disk = ['# N', '', ...CTR(5), '^a', '', 'p', '', ...CTR(5), '^b', '', 'END'].join('\n');
		const { app, host } = hovered(disk, { text: CTR(5).join('\n'), lineStart: 0, lineEnd: 3 }, 'Note#^a', bodyOf(5), { a: [10, 13], b: [10, 13] });
		const r = await plus(host, bodyOf(6));
		rec('T15 cache points ^a at B', `${r} ${JSON.stringify(vals(app))} notices ${Notice.notices.length}`);
		expect(vals(app)).toEqual([5, 5]);
	});
	test('T16 a callout counter under ## Tracker, an identical callout elsewhere already reading 6: +, then a second + with the window still stale: lands in A both times', async () => {
		const CAL = (n: number) => ['> [!note] Camp', ...CTR(n).map(q)];
		const disk = ['# N', '', 'TOP', '', ...CAL(6), '', 'p', '', '## Tracker', '', ...CAL(5), '', '## After', '', 'END'].join('\n');
		const part = ['', ...CAL(5), '', ''];
		const { app, host } = hovered(disk, { text: part.join('\n'), lineStart: 1, lineEnd: 5 }, 'Note#Tracker');
		const r1 = await plus(host, bodyOf(6));
		const r2 = await plus(host, bodyOf(7));
		rec('T16 callout, second + stale window', `${r1} ${r2} ${JSON.stringify(vals(app))} notices ${Notice.notices.length}`);
		expect(vals(app)).toEqual([6, 7]);
	});
