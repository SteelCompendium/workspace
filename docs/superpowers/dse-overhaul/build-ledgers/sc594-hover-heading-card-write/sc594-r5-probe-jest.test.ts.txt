// SC-594 r5 reviewer probes (UNCOMMITTED; copy kept in the ledger dir as sc594-r5-probe-jest.test.ts.txt). The popover DOM is drawn the way
// Obsidian 1.15.1 does (measured): `.markdown-embed[data-type=heading]` + `[data-heading]`, `data-type=block`, or neither (whole note).
import { ReadingModeBlockHost } from '../../../src/framework/host/ReadingModeBlockHost';
import { hoverLinkOf, noteHoverLink, popoverShows } from '../../../src/framework/host/hoverLinks';
import { atxHeadings } from '../../../src/framework/host/headingNames';
import { droppedWriteMessage, resetDroppedWriteNotices } from '../../../src/framework/host/droppedWriteNotice';
import { App, Notice, Plugin } from '../../mocks/obsidian';
import type { MarkdownPostProcessorContext } from '../../mocks/obsidian';

type Sec = { text: string; lineStart: number; lineEnd: number };
const ctxOf = (sec: Sec): MarkdownPostProcessorContext =>
	({ docId: 'doc-594', sourcePath: 'Note.md', frontmatter: undefined, addChild: () => {}, getSectionInfo: () => sec }) as never;
const CTR = (n: number): string[] => ['```ds-counter', 'name: Twin', `current_value: ${n}`, '```'];
const bodyOf = (n: number): string => `name: Twin\ncurrent_value: ${n}`;
const pos = (line: number) => ({ line, col: 0, offset: 0 });
const settle = () => new Promise((r) => setTimeout(r, 25));
function cacheFor(lines: string[], blocks: Record<string, [number, number]> = {}) {
	const headings: Array<{ heading: string; level: number; position: unknown }> = [];
	lines.forEach((l, i) => {
		const h = /^(#{1,6})\s+(.*)$/.exec(l);
		if (h) headings.push({ heading: h[2], level: h[1].length, position: { start: pos(i), end: pos(i) } });
	});
	const bl: Record<string, unknown> = {};
	for (const [id, [s, e]] of Object.entries(blocks)) bl[id] = { id, position: { start: pos(s), end: pos(e) } };
	return { headings, blocks: bl };
}
/** shows: 'heading:<name>' | 'block' | 'whole' */
function popoverEl(shows: string): { popover: HTMLElement; el: HTMLElement } {
	const popover = document.createElement('div');
	popover.className = 'popover hover-popover';
	const embed = popover.appendChild(document.createElement('div'));
	embed.className = 'markdown-embed';
	if (shows.startsWith('heading:')) {
		embed.setAttribute('data-type', 'heading');
		embed.appendChild(document.createElement('h2')).setAttribute('data-heading', shows.slice(8));
	} else if (shows === 'block') embed.setAttribute('data-type', 'block');
	const el = embed.appendChild(document.createElement('div'));
	document.body.appendChild(popover);
	return { popover, el };
}
type Parent = { hoverPopover: { hoverEl: HTMLElement } | null };
const ev = (linktext: string, parent: Parent) => ({ linktext, sourcePath: 'Host.md', hoverParent: parent });
const valuesOf = (app: App): number[] => [...app.vault.getContent('Note.md')!.matchAll(/current_value: (\d+)/g)].map((m) => Number(m[1]));

describe('SC-594 r5 probes', () => {
	let warn: jest.SpyInstance;
	const pops: HTMLElement[] = [];
	beforeEach(() => {
		resetDroppedWriteNotices();
		Notice.notices.length = 0;
		warn = jest.spyOn(console, 'warn').mockImplementation(() => {});
	});
	afterEach(() => {
		warn.mockRestore();
		pops.splice(0).forEach((p) => p.remove());
	});
	function mk(disk: string, sec: Sec, shows: string, cachedNote: string | null = null, blocks: Record<string, [number, number]> = {}) {
		const app = new App();
		app.vault.setFile('Note.md', disk);
		jest.spyOn(app.metadataCache, 'getFileCache').mockImplementation(() => cacheFor((cachedNote ?? app.vault.getContent('Note.md')!).split('\n'), blocks) as never);
		const { popover, el } = popoverEl(shows);
		pops.push(popover);
		const host = new ReadingModeBlockHost(new Plugin(app) as never, el, ctxOf(sec) as never, 'ds-counter');
		host.setMountedBody(bodyOf(5));
		return { app, host, popover, parent: { hoverPopover: null } as Parent };
	}
	async function plus(host: ReadingModeBlockHost): Promise<boolean> {
		host.notePersistIntent();
		await settle();
		return host.replaceSource(bodyOf(6), { fromView: true });
	}
	const TWO = ['# N', '', '## Tracker', '', ...CTR(5), '', '## Other', '', ...CTR(5), '', 'END'].join('\n');
	const TWOA = ['# N', '', 'TOP', '', '## Other', '', ...CTR(5), '', '## Tracker', '', ...CTR(5), '', 'END'].join('\n'); // twin above
	const heading: Sec = { text: ['', ...CTR(5), '', ''].join('\n'), lineStart: 1, lineEnd: 4 };
	const refused = (app: App, note: string) => {
		expect(app.vault.getContent('Note.md')).toBe(note);
		expect(Notice.notices).toEqual([droppedWriteMessage('Note')]);
	};

	// r3 probes P2 / P2b / P3 / P3b, with the preview DOM Obsidian draws for [[Note#Tracker]]
	test('P2 open before load (no own event) + pass-over [[Note#Other]]: refused', async () => {
		const { app, host, popover, parent } = mk(TWO, heading, 'heading:Tracker');
		parent.hoverPopover = { hoverEl: popover };
		noteHoverLink(app, ev('Note#Other', parent));
		expect(await plus(host)).toBe(false);
		refused(app, TWO);
	});
	test('P2b open before load + pass-over [[Note]] (whole note), twin above: refused', async () => {
		const { app, host, popover, parent } = mk(TWOA, heading, 'heading:Tracker');
		parent.hoverPopover = { hoverEl: popover };
		noteHoverLink(app, ev('Note', parent));
		expect(await plus(host)).toBe(false);
		refused(app, TWOA);
	});
	test('P3 residual heading: refused', async () => {
		const { app, host, popover, parent } = mk(TWO, heading, 'heading:Tracker');
		noteHoverLink(app, ev('Note#Tracker', parent));
		noteHoverLink(app, ev('Note#Other', parent));
		parent.hoverPopover = { hoverEl: popover };
		expect(await plus(host)).toBe(false);
		refused(app, TWO);
	});
	test('P3b residual whole-note Y, twin above: refused', async () => {
		const { app, host, popover, parent } = mk(TWOA, heading, 'heading:Tracker');
		noteHoverLink(app, ev('Note#Tracker', parent));
		noteHoverLink(app, ev('Note', parent));
		parent.hoverPopover = { hoverEl: popover };
		expect(await plus(host)).toBe(false);
		refused(app, TWOA);
	});
	// R1: the residual with two BLOCK links (both previews are data-type=block; the DOM cannot tell ^a from ^b)
	test('R1 residual with block links (^a preview bound to ^b): records whether the twin is written', async () => {
		const disk = ['# N', '', ...CTR(5), '^a', '', 'p', '', ...CTR(5), '^b', '', 'END'].join('\n');
		const { app, host, popover, parent } = mk(disk, { text: CTR(5).join('\n'), lineStart: 0, lineEnd: 3 }, 'block', null, { a: [2, 5], b: [10, 13] });
		noteHoverLink(app, ev('Note#^a', parent));
		noteHoverLink(app, ev('Note#^b', parent));
		parent.hoverPopover = { hoverEl: popover };
		await plus(host);
		expect(valuesOf(app)).toEqual([5, 5]); // a write into ^b ([5,6]) would be the twin
	});
	// R2: fresh cache, the hovered (first) `## Tracker` section deleted, another `## Tracker` with the identical counter left (measured in Obsidian: the
	// preview is NOT redrawn, its part text is unchanged; the cache catches up in 6-45 ms)
	test('R2 fresh cache, hovered first ## Tracker deleted, an identical ## Tracker section left: records whether it is written', async () => {
		const disk = ['# N', '', 'TOP', '', '## Mid', '', 'x', '', '## Tracker', '', ...CTR(5), '', 'END'].join('\n');
		const { app, host, popover, parent } = mk(disk, heading, 'heading:Tracker');
		parent.hoverPopover = { hoverEl: popover };
		noteHoverLink(app, ev('Note#Tracker', parent));
		await plus(host);
		expect(valuesOf(app)).toEqual([5]);
	});
	// R3: fresh cache, the two heading names swapped from outside
	test('R3 fresh cache, heading names swapped (hovered counter now under ## Other): records which is written', async () => {
		const disk = ['# N', '', '## Other', '', ...CTR(5), '', '## Tracker', '', ...CTR(5), '', 'END'].join('\n');
		const { app, host, popover, parent } = mk(disk, heading, 'heading:Tracker');
		parent.hoverPopover = { hoverEl: popover };
		noteHoverLink(app, ev('Note#Tracker', parent));
		await plus(host);
		expect(valuesOf(app)).not.toEqual([5, 6]);
	});
	// R4: no false refusals from the DOM check: nested subpath, heading with closing hashes, case / spacing differences
	test.each([
		['Note#Top#Tracker', 'heading:Tracker', ['# Top', '', '## Tracker', '', ...CTR(5), '', '## Other', '', ...CTR(5), '', 'END']],
		['Note#Tracker', 'heading:Tracker', ['# N', '', '## Tracker ##', '', ...CTR(5), '', '## Other', '', ...CTR(5), '', 'END']],
		['Note#tracker  log', 'heading:Tracker Log', ['# N', '', '## Tracker Log', '', ...CTR(5), '', '## Other', '', ...CTR(5), '', 'END']],
	] as const)('R4 %s with DOM %s lands in the hovered counter', async (link, shows, lines) => {
		const { app, host, popover, parent } = mk([...lines].join('\n'), heading, shows);
		parent.hoverPopover = { hoverEl: popover };
		noteHoverLink(app, ev(link, parent));
		expect(await plus(host)).toBe(true);
		expect(valuesOf(app)).toEqual([6, 5]);
	});
	test('R5 popoverShows / atxHeadings units', () => {
		const p = popoverEl('heading:Tracker').popover;
		pops.push(p);
		expect(popoverShows(p, 'Note#Tracker')).toBe(true);
		expect(popoverShows(p, 'Note#Other')).toBe(false);
		expect(popoverShows(p, 'Note')).toBe(false);
		expect(popoverShows(p, 'Note#^a')).toBe(false);
		expect(atxHeadings(['## A ##', '    ## code', '```', '## in', '```', '#NoSpace', '#', '### C'])).toEqual([
			{ line: 0, name: 'A' },
			{ line: 7, name: 'C' },
		]);
	});
	test('R6 hoverLinkOf with a correct bind but the DOM not yet drawn (a click before the heading renders) refuses, does not mis-bind', () => {
		const app = new App();
		const { popover, el } = popoverEl('heading:Tracker');
		pops.push(popover);
		noteHoverLink(app, ev('Note#Tracker', { hoverPopover: { hoverEl: popover } }));
		expect(hoverLinkOf(app, el)?.linktext).toBe('Note#Tracker');
	});
});
