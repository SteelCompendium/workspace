/** SC-594 probe: the `hover-link` events seen (the link text and the HoverParent whose popover shows it), newest last, bounded. */
type HoverLinkEvent = { linktext?: unknown; sourcePath?: unknown; hoverParent?: { hoverPopover?: { hoverEl?: HTMLElement } | null } | null };
const hoverLinks: HoverLinkEvent[] = [];
export function noteHoverLink(ev: HoverLinkEvent): void {
	if (!ev || typeof ev.linktext !== 'string') return;
	hoverLinks.push(ev);
	if (hoverLinks.length > 16) hoverLinks.shift();
}
export function resetHoverLinks(): void {
	hoverLinks.length = 0;
}
const hoverLinkApps = new WeakSet<object>();
/** SC-594 probe: the link of the hover preview `el` is drawn in: the newest `hover-link` event whose parent's popover IS that preview. */
function hoverLinkOf(el: HTMLElement): { linktext: string; sourcePath: string } | null {
	const pop = el.closest('.hover-popover');
	if (!pop) return null;
	for (let i = hoverLinks.length - 1; i >= 0; i--) {
		const ev = hoverLinks[i];
		if (ev.hoverParent?.hoverPopover?.hoverEl === pop) return { linktext: ev.linktext as string, sourcePath: typeof ev.sourcePath === 'string' ? ev.sourcePath : '' };
	}
	return null;
}
""")
