          const v = Math.round(parseFloat(el.style.minHeight));

          push('mh', where(el), v);

async function p472Run(s, t) {
	const thr = P472_THR > 1 && (!P472_THR_IDS.length || P472_THR_IDS.includes(s.id)) ? P472_THR : 1;
	const la0 = fs.readFileSync('/proc/loadavg', 'utf8').trim();
	await t.ev(P472_REC_INSTALL);
	if (thr > 1) await t.cdp.call('Emulation.setCPUThrottlingRate', { rate: thr });
	const w0 = Date.now();
	try {
		return await s.run(t);
	} finally {
		try {
			if (thr > 1) await t.cdp.call('Emulation.setCPUThrottlingRate', { rate: 1 });
		} catch {}
		let tl = null;
		try {
			tl = await t.ev('window.__p472 ? window.__p472.dump() : null');
		} catch (e) {
			tl = JSON.stringify('dump failed ' + e);
		}
		console.log(`P472-TL ${s.id} thr=${thr} wall=${Date.now() - w0}ms load0=[${la0}] load1=[${fs.readFileSync('/proc/loadavg', 'utf8').trim()}] ${tl}`);
	}
}
