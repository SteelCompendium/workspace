import sys
# CONTROL: the backstop's busy rules removed everywhere (frames + quiet frames + 20 s cap only)
t = sys.argv[1]
def rep(rel, old, new):
    p = t + '/' + rel
    s = open(p).read()
    assert s.count(old) == 1, (rel, old[:50], s.count(old))
    open(p, 'w').write(s.replace(old, new))

    "(quiet) => quiet < 0 && BACKSTOP_REDRAW_QUIET_FRAMES > 0 && emptyRoot(scroller)); // CONTROL nobusy")
