// Page side: DSE-origin rAF / setTimeout / MutationObserver accounting (stack contains 'draw-steel-elements'), and capture of the pin / hold Maps.
function r3PageInstall() {
	if (window.__r3) return 'already';
	const st = { raf: 0, rafMs: 0, rafMax: 0, active: new Set(), tmo: 0, tmoActive: new Set(), sample: null, moObs: 0, moDisc: 0, pins: null, holds: null };
	window.__r3 = st;
	const isDse = () => {
		const s = new Error().stack || '';
		const hit = s.includes('draw-steel-elements');
		if (hit && !st.sample) st.sample = s.slice(0, 500);
		return hit;
	};
	const oRaf = window.requestAnimationFrame.bind(window);
	const oCaf = window.cancelAnimationFrame.bind(window);
	st.oRaf = oRaf;
	window.requestAnimationFrame = function (cb) {
		if (!isDse()) return oRaf(cb);
		st.raf++;
		const id = oRaf((ts) => {
			st.active.delete(id);
			const a = performance.now();
			try {
				cb(ts);
			} finally {
				const d = performance.now() - a;
				st.rafMs += d;
				if (d > st.rafMax) st.rafMax = d;
			}
		});
		st.active.add(id);
		return id;
	};
	window.cancelAnimationFrame = function (id) {
		st.active.delete(id);
		return oCaf(id);
	};
	const oSt = window.setTimeout.bind(window);
	const oCt = window.clearTimeout.bind(window);
	window.setTimeout = function (cb, ms, ...a) {
		if (typeof cb !== 'function' || !isDse()) return oSt(cb, ms, ...a);
		st.tmo++;
		const id = oSt(() => {
			st.tmoActive.delete(id);
			cb(...a);
		}, ms);
		st.tmoActive.add(id);
		return id;
	};
	window.clearTimeout = function (id) {
		st.tmoActive.delete(id);
		return oCt(id);
	};
	const OMO = window.MutationObserver;
	st.OMO = OMO;
	class MO extends OMO {
		constructor(cb) {
			super(cb);
			this.__dse = isDse();
			this.__on = false;
		}
		observe(...a) {
			if (this.__dse && !this.__on) {
				st.moObs++;
				this.__on = true;
			}
			return super.observe(...a);
		}
		disconnect() {
			if (this.__dse && this.__on) {
				st.moDisc++;
				this.__on = false;
			}
			return super.disconnect();
		}
	}
	window.MutationObserver = MO;
	const oSet = Map.prototype.set;
	Map.prototype.set = function (k, v) {
		try {
			if (v && typeof v === 'object' && 'px' in v && 'style' in v && 'id' in v && 'win' in v) st.pins = this;
			if (typeof k === 'function' && v && typeof v === 'object' && 'contentDOM' in v) st.holds = this;
		} catch {
			/* probe */
		}
		return oSet.call(this, k, v);
	};
	return 'installed';
}
const r3Stats = `(() => { const s = window.__r3; return { raf: s.raf, rafMs: Math.round(s.rafMs * 100) / 100, rafMax: Math.round(s.rafMax * 100) / 100, rafActive: s.active.size, tmo: s.tmo, tmoActive: s.tmoActive.size, moObs: s.moObs, moDisc: s.moDisc, moNet: s.moObs - s.moDisc, pins: s.pins ? s.pins.size : null, holds: s.holds ? s.holds.size : null, pinAttr: document.querySelectorAll('[data-dse-scroll-pin]').length, pinStyle: document.querySelectorAll('style[data-dse-scroll-pin-style]').length, lpHold: document.querySelectorAll('[data-dse-lp-hold]').length, embHold: document.querySelectorAll('[data-dse-lp-embed-hold]').length }; })()`;

// Page side: a per-frame recorder of one scroller (Reading `.markdown-preview-view` or `.cm-scroller`), compacted to changes.
function r3PageRec(s, tagged, opts) {
	const T0 = performance.now();
	window.__r3t0 = T0;
	const ev = [];
	const last = {};
	const now = () => Math.round(performance.now() - T0);
	const push = (k, v) => {
		if (last[k] === v) return;
		last[k] = v;
		if (ev.length < 1500) ev.push([now(), k, v]);
	};
	const isLp = s.classList.contains('cm-scroller');
	const rec = { ev, frames: 0, st0: Math.round(s.scrollTop), sh0: s.scrollHeight, ch: s.clientHeight, stMin: Infinity, stMax: -Infinity, done: false, mods: [], tagOff: null, blankFrames: 0, blankMax: 0 };
	const attrs = opts.attrs || ['data-dse-scroll-pin'];
	const OMO = (window.__r3 && window.__r3.OMO) || MutationObserver;
	const mo = new OMO((recs) => {
		for (const r of recs) {
			if (r.type === 'attributes' && attrs.includes(r.attributeName)) {
				const on = r.target.hasAttribute(r.attributeName);
				push('A:' + r.attributeName + (r.target === s ? '' : '@' + String(r.target.className).slice(0, 24)), on ? 1 : 0);
			}
		}
		if (tagged && !tagged.isConnected && rec.tagOff === null) rec.tagOff = now();
	});
	mo.observe(s, { subtree: true, childList: true, attributes: true, attributeFilter: attrs });
	const ref = app.vault.on('modify', (f) => {
		rec.mods.push([now(), f.path]);
	});
	const raf = (window.__r3 && window.__r3.oRaf) || window.requestAnimationFrame.bind(window);
	const finish = () => {
		if (rec.done) return;
		rec.done = true;
		mo.disconnect();
		app.vault.offref(ref);
		rec.stEnd = Math.round(s.scrollTop);
		rec.shEnd = s.scrollHeight;
	};
	const tick = () => {
		if (rec.done) return;
		rec.frames++;
		const v = Math.round(s.scrollTop);
		if (v < rec.stMin) rec.stMin = v;
		if (v > rec.stMax) rec.stMax = v;
		push('st', v);
		push('sh', s.scrollHeight);
		if (!isLp) {
			const z = s.querySelector(':scope > .markdown-preview-sizer');
			push('mh', z ? Math.round(parseFloat(z.style.minHeight)) : null);
			push('pin', s.hasAttribute('data-dse-scroll-pin') ? 1 : 0);
			if (opts.blank && z) {
				// blank band: viewport bottom below the last rendered block's bottom
				const kids = Array.from(z.children).filter((c) => c.getBoundingClientRect().height > 0);
				const lastKid = kids[kids.length - 1];
				const vb = s.getBoundingClientRect().bottom;
				const gap = lastKid ? Math.round(vb - lastKid.getBoundingClientRect().bottom) : null;
				push('gap', gap);
				if (gap !== null && gap > 80) {
					rec.blankFrames++;
					rec.blankMax = Math.max(rec.blankMax, gap);
					if (rec.blankFirst === undefined) rec.blankFirst = now();
					rec.blankLast = now();
				}
			}
		}
		const rs = Array.from(s.querySelectorAll('[data-dse-element]'));
		push('roots', rs.length + '/' + rs.filter((r) => r.childElementCount === 0).length);
		if (tagged) push('tag', tagged.isConnected ? 1 : 0);
		if (opts.lpb) {
			// SC-478 pane: holds and blank (connected, unmounted) DSE widgets
			const ws = Array.from(s.querySelectorAll('.cm-content > .cm-embed-block')).filter((w) => /cm-lang-ds-/.test(w.className));
			const blank = ws.filter((w) => {
				const r = w.querySelector('[data-dse-element]');
				return !r || r.childElementCount === 0;
			});
			const vis = blank.filter((w) => {
				const b = w.getBoundingClientRect();
				const sr = s.getBoundingClientRect();
				return b.bottom > sr.top && b.top < sr.bottom && b.height > 0;
			});
			push('lhold', s.querySelectorAll('[data-dse-lp-hold]').length);
			push('blankW', blank.length + '/' + vis.length);
			if (vis.length) {
				rec.blankFrames++;
				if (rec.blankFirst === undefined) rec.blankFirst = now();
				rec.blankLast = now();
			}
		}
		if (performance.now() - T0 > (opts.ms || 10000)) return finish();
		raf(tick);
	};
	rec.stop = finish;
	window.__r3rec = rec;
	raf(tick);
	return true;
}
const r3Rec = (t, scrollerExpr, taggedExpr, opts) => t.ev(`(${r3PageRec.toString()})(${scrollerExpr}, ${taggedExpr}, ${JSON.stringify(opts)})`);
const r3Install = (t) => t.ev(`(${r3PageInstall.toString()})()`);
/** events of kind k after time a */
const r3Pick = (rec, k, a = -1) => rec.ev.filter((e) => e[1] === k && e[0] >= a);
const r3PinOnOff = (rec, k = 'pin') => {
	const on = rec.ev.find((e) => e[1] === k && e[2] === 1);
	const off = on ? rec.ev.find((e) => e[1] === k && e[2] === 0 && e[0] > on[0]) : null;
	return { on: on ? on[0] : null, off: off ? off[0] : null };
};
