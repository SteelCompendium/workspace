    assert s.count(old) == 1, (old[:50], s.count(old))
    s = s.replace(old, new)

    rep("\t\tthis.cancelCheck(held); // a check queued by the redraw that began before this arm must not judge the new one\n", "\t\t// CONTROL nodev-i\n")
