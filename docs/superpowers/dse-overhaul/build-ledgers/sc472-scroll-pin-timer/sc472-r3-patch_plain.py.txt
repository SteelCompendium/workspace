# CONTROL: release on the redraw + a plain 2.5 s timer (the backstop fires at the deadline, no frame count / busy rule)
p = sys.argv[1] + '/src/framework/host/redrawBackstop.ts'

new = "this.timer = null;\n\t\treturn this.done(); // CONTROL: plain timer\n\t\tthis.frames = 0;"
assert s.count(old) == 1, s.count(old)
