#!/bin/bash
# SC-525 r3: (a) the new SCC re-entrance test against af09bf2's source (must be red); (b) mutants of the r3 head.
# Each mutant edits RefUnwrapView.ts, runs the three suites, restores the file from git.
P=/home/scott/code/steelCompendium/worktrees/sc525-ref-nesting-retype/draw-steel-elements
D=/home/scott/code/steelCompendium/workspace/.superpowers/sdd/sc525-ref-nesting-retype
OUT=$D/sc525-r3-mutants.log
cd "$P" || exit 9
SUITES="test/dom/elements/refUnwrapView.test.ts test/dom/authoring/formModalStalePreview.test.ts test/dom/elements/displayCardHybrid.test.ts"
R=src/elements/shared/RefUnwrapView.ts
: > "$OUT"

run_suites() {
	rm -f main.js styles.css
	npx jest $SUITES --silent > "$OUT.tmp" 2>&1
	grep -E "^\s+●.*›|^Tests:" "$OUT.tmp" | sort -u >> "$OUT"
}

echo "=== BASE-OF-R3 (af09bf2 source, r3 tests)" >> "$OUT"
git show af09bf2:$R > "$R"
run_suites
git checkout -- "$R"

run_mutant() {
	name="$1"; from="$2"; to="$3"
	python3 - "$R" "$from" "$to" <<'PY'
import sys
p, a, b = sys.argv[1:4]
s = open(p).read()
assert s.count(a) == 1, 'pattern count %d' % s.count(a)
open(p, 'w').write(s.replace(a, b))
PY
	echo "=== MUTANT $name (applied rc: $?)" >> "$OUT"
	run_suites
	git checkout -- "$R"
}

# drop gen checks; $1 = which: all | legacy | scc
drop_gen() {
	name="$1"; which="$2"
	python3 - "$R" "$which" <<'PY'
import sys, re
p, which = sys.argv[1:3]
lines = open(p).read().split('\n')
out = []
section = None
for ln in lines:
    if 'private async resolveLegacyRef' in ln: section = 'legacy'
    elif 'private async resolveAndMount' in ln: section = 'scc'
    if 'if (gen !== this.resolveGen)' in ln and which in ('all', section):
        if '{' in ln or ln.strip().endswith('return;') and 'catch' in ln:
            pass
        ln = ln.split('if (gen !== this.resolveGen)')[0] + '/* gen check dropped */'
    out.append(ln)
open(p, 'w').write('\n'.join(out))
PY
	echo "=== MUTANT $name (gen checks dropped: $which; diff: $(git diff --numstat -- $R | cut -f1,2 | tr '\t' '/'))" >> "$OUT"
	run_suites
	git checkout -- "$R"
}

run_mutant "M1-owner-unloaded-check-removed" "current.view !== this && !current.view.isUnloaded()" "current.view !== this"
run_mutant "M2-delete-unconditionally" "if (IN_FLIGHT_REFS.get(guardKey) === token) IN_FLIGHT_REFS.delete(guardKey);" "IN_FLIGHT_REFS.delete(guardKey);"
run_mutant "M4-same-view-clause-dropped (same view re-entrance refused again)" "current !== undefined && current.view !== this && !current.view.isUnloaded()" "current !== undefined && !current.view.isUnloaded()"
run_mutant "M5-no-takeover (set only when no entry)" "IN_FLIGHT_REFS.set(guardKey, token);" "if (!IN_FLIGHT_REFS.has(guardKey)) IN_FLIGHT_REFS.set(guardKey, token);"
run_mutant "M7-delete-by-view (token compare replaced by view compare)" "if (IN_FLIGHT_REFS.get(guardKey) === token) IN_FLIGHT_REFS.delete(guardKey);" "if (IN_FLIGHT_REFS.get(guardKey)?.view === this) IN_FLIGHT_REFS.delete(guardKey);"
drop_gen "M6-all-resolveGen-checks-dropped" all
drop_gen "M6a-legacy-resolveGen-checks-dropped" legacy
drop_gen "M6b-scc-resolveGen-checks-dropped" scc
rm -f "$OUT.tmp"
echo "=== restored: $(git status --porcelain | wc -l) changed files (the uncommitted SCC test is expected: 1)" >> "$OUT"
exit 0
