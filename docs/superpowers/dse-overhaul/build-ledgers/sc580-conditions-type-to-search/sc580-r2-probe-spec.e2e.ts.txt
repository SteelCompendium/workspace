/** Fire key events through CDP without awaiting each one: a no-pause burst. */
async function burst(text: string): Promise<void> {
	const sends: Promise<unknown>[] = [];
	for (const ch of text) {
		const code = /[a-z]/i.test(ch) ? `Key${ch.toUpperCase()}` : '';
		const vk = ch.toUpperCase().charCodeAt(0);
		sends.push(browser.sendCommandAndGetResult('Input.dispatchKeyEvent', { type: 'keyDown', key: ch, code, text: ch, unmodifiedText: ch, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk }));
		sends.push(browser.sendCommandAndGetResult('Input.dispatchKeyEvent', { type: 'keyUp', key: ch, code, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk }));
	}
	await Promise.all(sends);
}
