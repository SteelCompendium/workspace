// sc548 r1: extract one file from an asar archive. usage: node sc548-r1-asarget.cjs <asar> <inner path> <out>
const fs = require('fs');
const [asar, inner, out] = process.argv.slice(2);
const fd = fs.openSync(asar, 'r');
const b = Buffer.alloc(16);
fs.readSync(fd, b, 0, 16, 0);
const headerSize = b.readUInt32LE(4); // pickle: [4 size of pickle1][4 headerPickleSize] ... then [4 payload][4 strlen][json]
const jsonLen = b.readUInt32LE(12);
const hb = Buffer.alloc(jsonLen);
fs.readSync(fd, hb, 0, jsonLen, 16);
const header = JSON.parse(hb.toString('utf8'));
const base = 8 + headerSize;
let node = header;
for (const part of inner.split('/')) node = node.files[part];
const data = Buffer.alloc(node.size);
fs.readSync(fd, data, 0, node.size, base + Number(node.offset));
fs.writeFileSync(out, data);
console.log(`wrote ${out} ${node.size} bytes`);
