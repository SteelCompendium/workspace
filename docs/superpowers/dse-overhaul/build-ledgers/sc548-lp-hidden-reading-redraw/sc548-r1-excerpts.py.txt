import sys

new, v144, v137 = sys.argv[1:4]
s = open(new).read()


def show(label, anchor, before=0, after=600, src=s):
    i = src.find(anchor)
    assert i >= 0, label
    print(f'==== {label} (offset {i})')
    print(src[max(0, i - before):i + after])
    print()
