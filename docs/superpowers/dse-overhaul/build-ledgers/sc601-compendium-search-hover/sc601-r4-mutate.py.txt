def run(tag):
    p = subprocess.run(['npx', 'jest', *TESTS], cwd=ROOT, capture_output=True, text=True)
    out = p.stdout + p.stderr
    open(f'{S}/mut4-{tag}.log', 'w').write(out)
    tests = [l for l in out.splitlines() if l.startswith('Tests:')]
    failed = sorted({l.strip()[2:] for l in out.splitlines() if l.startswith('  ● ')})
    return p.returncode, (tests[0] if tests else '?'), failed


def main():
    only = set(sys.argv[1:])
    rows = []
    for name, rel, edits in MUTANTS:
        tag = name.split()[0]
        if only and tag not in only:
            continue
        path = os.path.join(ROOT, rel)
        orig = open(path).read()
        new = orig
        for a, b in edits:
            if a not in new:
                rows.append(f'{name}: NOT APPLIED (pattern missing)')
                new = None
                break
            new = new.replace(a, b)
        if new is None:
            continue
        open(path, 'w').write(new)
        try:
            rc, tests, failed = run(tag)
        finally:
            open(path, 'w').write(orig)
        verdict = 'KILLED' if rc != 0 else 'SURVIVED'
        rows.append(f'{name}: {verdict} ({tests}); failing: {failed[:4]}{" ..." if len(failed) > 4 else ""}')
        print(rows[-1], flush=True)
    open(f'{S}/mutants4-summary.txt', 'a').write('\n'.join(rows) + '\n')


main()
