const pos = (line: number) => ({ line, col: 0, offset: 0 });
function cacheOf(lines: string[]) {
	const headings: Array<{ heading: string; level: number; position: unknown }> = [];
	const blocks: Record<string, unknown> = {};
	let fenced = false;
	lines.forEach((l, i) => {
		if (/^```/.test(l)) fenced = !fenced;
		const h = !fenced && /^(#{1,6})\s+(.*)$/.exec(l);
		if (h) headings.push({ heading: h[2], level: h[1].length, position: { start: pos(i), end: pos(i) } });
		const id = /^\^(\w+)$/.exec(l);
		if (id) {
			let s = i - 1;
			while (s >= 0 && lines[s] !== '```') s--;
			for (s = s - 1; s >= 0 && !/^```\w/.test(lines[s]); s--);
			blocks[id[1]] = { id: id[1], position: { start: pos(s), end: pos(i - 1) } };
		}
	});
	// a last section ending where the text ends (metaMatchesText reads it)
	const sections = [{ type: 'paragraph', position: { start: pos(0), end: { line: lines.length - 1, col: lines[lines.length - 1].length, offset: lines.join('\n').length } } }];
	return { headings, blocks, sections };
}
function popoverEl(app: App, link: string, data?: unknown): HTMLElement {
	const hoverEl = document.createElement('div');
	hoverEl.className = 'popover hover-popover';
	const embed = hoverEl.appendChild(document.createElement('div'));
	embed.className = 'markdown-embed';
	const sub = link.includes('#') ? link.slice(link.indexOf('#')) : null;
	if (sub === null) {
		// a whole note: neither data-type
	} else if (sub.startsWith('#^')) embed.setAttribute('data-type', 'block');
	else {
		embed.setAttribute('data-type', 'heading');
		const h = embed.appendChild(document.createElement('h2'));
		h.setAttribute('data-heading', sub!.split('#').pop()!);
	}
	const el = embed.appendChild(document.createElement('div'));
	document.body.appendChild(hoverEl);
	const popover = data === undefined ? { hoverEl } : { hoverEl, embed: { data } };
	noteHoverLink(app, { linktext: link, sourcePath: 'Host.md', hoverParent: { hoverPopover: popover } });
	return el;
}
function embedEl(src: string): HTMLElement {
	const embed = document.createElement('div');
	embed.className = 'internal-embed markdown-embed';
	embed.setAttribute('src', src);
	const el = embed.appendChild(document.createElement('div'));
	document.body.appendChild(embed);
	return el;
}
const fenceStarts = (lines: string[]): number[] => lines.flatMap((l, i) => (l.startsWith('```ds-counter') ? [i] : []));
const noticeKind = (n: string): string => (n.includes('changed on disk first') ? 'dropped' : n.includes("can't tell where") ? 'unsavable' : 'other');

type Step =
	| { do: 'outside'; lines: string[]; cache?: 'fresh' | 'lagging' }
	| { do: 'cache' } // the cache catches up
	| { do: 'formopen' } // the pencil: the form's pin is taken (and rebased) now
	| { do: 'formsave' } // Save of the form opened earlier
	| { do: 'click'; card: 'card' | 'form'; value?: number }
	| { do: 'redraw'; sec: Sec; via: 'rebind' | 'mount'; data?: unknown; known?: number };
type Probe = {
	name: string;
	shape: 'hover' | 'embed' | 'plain';
	link: string;
	drawn: string[]; // the note Obsidian drew the part from (embed.data)
	snapshot?: string[]; // what the card's mount read sees (default drawn)
	snapshotCache?: string[]; // the cache at the mount read (default snapshot)
	data?: unknown; // embed.data (default drawn text; 'absent' = no embed object)
	sec: Sec;
	known: number;
	steps: Step[];
	// the fence line (in the FINAL note) of the counter the card was drawn from, or -1 if it is gone; the probe judges every write against it
	drawnFromNow: number;
};

async function runProbe(p: Probe): Promise<{ outcome: string; wrote: number[]; notices: string[] }> {
	const app = new App();
	const snap = p.snapshot ?? p.drawn;
	app.vault.setFile('Note.md', snap.join('\n'));
	let cacheLines = p.snapshotCache ?? snap;
	jest.spyOn(app.metadataCache, 'getFileCache').mockImplementation(() => cacheOf(cacheLines) as never);
	const section: Section = { current: p.sec };
	const data = p.data === 'absent' ? undefined : p.data === undefined ? p.drawn.join('\n') : p.data;
	const mk = (): HTMLElement => (p.shape === 'hover' ? popoverEl(app, p.link, data) : p.shape === 'plain' ? document.body.appendChild(document.createElement('div')) : embedEl(p.link));
	let el = mk();
	let host = new ReadingModeBlockHost(new Plugin(app) as never, el, ctxOf(section), 'ds-counter');
	host.setMountedBody(bodyOf(p.known));
	await settle();
	if (p.snapshot) {
		// the snapshot read is done; the note now is what the snapshot saw (the outside edit landed before it)
	}
	let known = p.known;
	let lastBefore = app.vault.getContent('Note.md')!;
	let wrote: number[] = [];
	let formPin: ReturnType<ReadingModeBlockHost['pinPosition']> = null;
	for (const s of p.steps) {
		if (s.do === 'outside') {
			app.vault.setFile('Note.md', s.lines.join('\n'));
			if (s.cache !== 'lagging') cacheLines = s.lines;
		} else if (s.do === 'formopen') {
			formPin = host.pinPosition();
			await settle();
		} else if (s.do === 'formsave') {
			Notice.notices.length = 0;
			resetDroppedWriteNotices();
			lastBefore = app.vault.getContent('Note.md')!;
			const ok = await (formPin ? host.replaceSource(bodyOf(known, 'Edited'), { pin: formPin, pinOnly: true }) : host.replaceSource(bodyOf(known, 'Edited')));
			const after = app.vault.getContent('Note.md')!;
			const bl = lastBefore.split('\n');
			const al = after.split('\n');
			wrote = after === lastBefore ? [] : fenceStarts(bl).filter((f) => bl.slice(f, f + 4).join('\n') !== al.slice(f, f + 4).join('\n'));
			if (ok) cacheLines = al;
		} else if (s.do === 'cache') {
			cacheLines = app.vault.getContent('Note.md')!.split('\n');
		} else if (s.do === 'redraw') {
			if (s.known !== undefined) known = s.known;
			section.current = s.sec;
			if (s.via === 'rebind') host.rebind(el, ctxOf(section));
			else {
				el.closest('.hover-popover, .internal-embed')?.remove();
				el = p.shape === 'hover' ? popoverEl(app, p.link, s.data === undefined ? data : s.data) : embedEl(p.link);
				host = new ReadingModeBlockHost(new Plugin(app) as never, el, ctxOf(section), 'ds-counter');
				host.setMountedBody(bodyOf(known));
			}
			await settle();
		} else {
			Notice.notices.length = 0;
			resetDroppedWriteNotices();
			lastBefore = app.vault.getContent('Note.md')!;
			let ok: boolean;
			if (s.card === 'card') {
				host.notePersistIntent();
				await settle();
				ok = await host.replaceSource(bodyOf(s.value ?? known + 1), { fromView: true });
			} else {
				const pin = host.pinPosition();
				ok = await (pin ? host.replaceSource(bodyOf(known, 'Edited'), { pin, pinOnly: true }) : host.replaceSource(bodyOf(known, 'Edited')));
			}
			const after = app.vault.getContent('Note.md')!;
			const bl = lastBefore.split('\n');
			const al = after.split('\n');
			wrote = after === lastBefore ? [] : fenceStarts(bl).filter((f) => bl.slice(f, f + 4).join('\n') !== al.slice(f, f + 4).join('\n'));
			if (ok && s.card === 'card') known = s.value ?? known + 1;
			if (ok) {
				cacheLines = al;
			}
		}
	}
	el.closest('.hover-popover, .internal-embed')?.remove();
	const notices = (Notice.notices as unknown[]).map((n) => noticeKind(String(n)));
	const outcome = wrote.length === 0 ? 'refused' : wrote.length === 1 && wrote[0] === p.drawnFromNow ? 'SAME' : 'DIFFERENT';
	return { outcome, wrote, notices };
}

	const k2 = k1.map((l) => (l === '^pb' ? '^pa' : l));
	if (shape === 'hover') probes.push({ name: `P11b ${shape} give-up N5 + marker renamed: ^pa deleted, ^pb edited and renamed ^pa, 1000 lines`, shape, link: 'Note#^pa', drawn: k0, sec: blkSec(5), known: 5, steps: [{ do: 'outside', lines: k2 }, { do: 'click', card: 'card' }], drawnFromNow: -1 });
	// P12 forms: the form opened, then the names swapped, then Save (form-before)
	probes.push({ name: `P12 ${shape} form pin taken, names swapped, Save`, shape, link: 'Note#Tracker', drawn: sw, sec, known: 5, steps: [{ do: 'outside', lines: swapped }, { do: 'click', card: 'form' }], drawnFromNow: at(swapped, 0) });
	// P13 the cache lags at the click (link resolves by the old cache), swap
	probes.push({ name: `P13 ${shape} names swapped, cache lagging at the click`, shape, link: 'Note#Tracker', drawn: sw, sec, known: 5, steps: [{ do: 'outside', lines: swapped, cache: 'lagging' }, { do: 'click', card: 'card' }], drawnFromNow: at(swapped, 0) });
	// P14 no edit: 3 + in a row with rebind redraws (no regression)
	probes.push({
		name: `P14 ${shape} no edit, + rebind + rebind +`, shape, link: 'Note#Tracker', drawn: sw, sec, known: 5,
		steps: [
			{ do: 'click', card: 'card' }, { do: 'redraw', sec: shape === 'hover' ? hoverSec(6) : embedSec('Tracker', 6), via: 'rebind' },
			{ do: 'click', card: 'card' }, { do: 'redraw', sec: shape === 'hover' ? hoverSec(7) : embedSec('Tracker', 7), via: 'rebind' },
			{ do: 'click', card: 'card' },
		],
		drawnFromNow: at(sw, 0),
	});
	// P15 no edit: + then a NEW-card redraw (a mount, popover data still the first-drawn note) then + (residual (2) of the amended guard when the new part is in the old data)
	probes.push({
		name: `P15 ${shape} no edit, + then new-card redraw (data stale) then +; Other=6 so the new part is in the stale data`, shape, link: 'Note#Tracker', drawn: note(H('Tracker', 5), H('Other', 6)), sec, known: 5,
		steps: [{ do: 'click', card: 'card' }, { do: 'redraw', sec: shape === 'hover' ? hoverSec(6) : embedSec('Tracker', 6), via: 'mount' }, { do: 'click', card: 'card' }],
		drawnFromNow: at(note(H('Tracker', 5), H('Other', 6)), 0),
	});
}

// P16-P18: whole-note hover, whole-note embed, plain card; edits none / unrelated / identical block added above / the twin deleted / the card's own block deleted
{
	const W0 = note(H('Tracker', 5), H('Other', 5));
	const Wsec = (lines: string[], k: number): Sec => ({ text: lines.join('\n'), lineStart: fenceStarts(lines)[k], lineEnd: fenceStarts(lines)[k] + 3 });
	const edits: Array<[string, string[], number]> = [
		['none', W0, at(W0, 0)],
		['unrelated', ['# N', '', 'TOP edited', '', ...H('Tracker', 5), ...H('Other', 5), '## After', '', 'END'], 6],
		['identical added above', ['# N', '', 'TOP', '', ...CTR(5), '', ...H('Tracker', 5), ...H('Other', 5), '## After', '', 'END'], 11],
		['the other deleted', note(H('Tracker', 5)), 6],
		['names swapped', note(H('Other', 5), H('Tracker', 5)), 6],

	for (const [shape, link] of [['hover', 'Note'], ['embed', 'Note'], ['plain', 'Note']] as const)

}
