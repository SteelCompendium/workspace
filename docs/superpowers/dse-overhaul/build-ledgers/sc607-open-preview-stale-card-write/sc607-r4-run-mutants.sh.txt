#!/usr/bin/env bash
# r2-run-mutants.sh <id>...: per mutant restore the 3 src files to the committed state, apply it, run the SC-607 suites, record into mut/results.log; restore at the end.
S=/tmp/claude-1000/-home-scott-code-steelCompendium-workspace/be71a0a9-199f-4e14-967b-191935312b05/scratchpad/sc607-r4-impl
E=/home/scott/code/steelCompendium/worktrees/sc607-open-preview-stale-card-write/draw-steel-elements
cd "$E" || exit 9
rm -f main.js styles.css
for m in "$@"; do
  git -C "$E" checkout -- src/framework/host/ReadingModeBlockHost.ts src/framework/host/lineMap.ts src/framework/host/hoverLinks.ts
  python3 -I "$S/r4-mutate.py" "$m" >> "$S/mut/results.log" 2>&1 || { echo "$m mutate_failed" >> "$S/mut/results.log"; continue; }
  npx jest test/dom/framework/reading-mode-host-drawn-pin.test.ts test/dom/framework/reading-mode-host-pin.test.ts test/dom/framework/reading-mode-host-hover-card.test.ts test/dom/framework/reading-mode-host-form-pin.test.ts test/unit/framework/lineMap.test.ts --silent > "$S/mut/mut-$m.log" 2>&1
  rc=$?
  echo "$m jest_rc=$rc $(grep -E '^Tests:' "$S/mut/mut-$m.log")" >> "$S/mut/results.log"
done
git -C "$E" checkout -- src/framework/host/ReadingModeBlockHost.ts src/framework/host/lineMap.ts src/framework/host/hoverLinks.ts
